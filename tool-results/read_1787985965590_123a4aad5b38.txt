   197→   197→/**
   198→   198→ * Project inquiry form with honeypot spam protection.
   199→   199→ * Posts to /api/leads (stored in DB) and fires a GA4 generate_lead event.
   200→   200→ */
   201→   201→export function LeadForm({
   202→   202→  source = 'contact',
   203→   203→  defaultService,
   204→   204→  dark,
   205→   205→  compact,
   206→   206→  bare,
   207→   207→  className,
   208→   208→}: LeadFormProps) {
   209→   209→  const { toast } = useToast();
   210→   210→  const [form, setForm] = React.useState<FormState>(() => ({
   211→   211→    ...EMPTY_FORM,
   212→   212→    service: defaultService ?? '',
   213→   213→  }));
   214→   214→  const [honeypot, setHoneypot] = React.useState('');
   215→   215→  const [errors, setErrors] = React.useState<Errors>({});
   216→   216→  const [submitting, setSubmitting] = React.useState(false);
   217→   217→  const [submitted, setSubmitted] = React.useState(false);
   218→   218→  const [leadId, setLeadId] = React.useState<string | null>(null);
   219→   219→
   220→   220→  /** Map form fields to their validation error keys. */
   221→   221→  const ERROR_KEY_FOR: Partial<Record<keyof FormState, keyof Errors>> = {
   222→   222→    firstName: 'name',
   223→   223→    email: 'email',
   224→   224→    service: 'service',
   225→   225→    budget: 'budget',
   226→   226→    message: 'message',
   227→   227→  };
   228→   228→
   229→   229→  const set = (key: keyof FormState) => (value: string) => {
   230→   230→    setForm((prev) => ({ ...prev, [key]: value }));
   231→   231→    const errorKey = ERROR_KEY_FOR[key];
   232→   232→    if (!errorKey) return;
   233→   233→    setErrors((prev) => (prev[errorKey] ? { ...prev, [errorKey]: undefined } : prev));
   234→   234→  };
   235→   235→
   236→   236→  const validate = (): boolean => {
   237→   237→    const next: Errors = {};
   238→   238→    if (form.firstName.trim().length < 2) next.name = 'Please enter your first name.';
   239→   239→    if (!EMAIL_REGEX.test(form.email.trim())) next.email = 'Please enter a valid email address.';
   240→   240→    if (!form.service) next.service = 'Please select a service.';
   241→   241→    if (!form.budget) next.budget = 'Please select a budget range.';
   242→   242→    if (form.message.trim().length < MESSAGE_MIN) {
   243→   243→      next.message = `Please add a little more detail (at least ${MESSAGE_MIN} characters).`;
   244→   244→    }
   245→   245→    setErrors(next);
   246→   246→    return Object.keys(next).length === 0;
   247→   247→  };
   248→   248→
   249→   249→  const handleSubmit = async (event: React.FormEvent) => {
   250→   250→    event.preventDefault();
   251→   251→    if (!validate()) return;
   252→   252→    setSubmitting(true);
   253→   253→    try {
   254→   254→      const fullName = [form.firstName.trim(), form.lastName.trim()].filter(Boolean).join(' ');
   255→   255→      // Company is folded into the message so nothing the client shares gets lost.
   256→   256→      const message = form.company.trim()
   257→   257→        ? `Company: ${form.company.trim()}\n\n${form.message}`
   258→   258→        : form.message;
   259→   259→
   260→   260→      const response = await fetch('/api/leads', {
   261→   261→        method: 'POST',
   262→   262→        headers: { 'Content-Type': 'application/json' },
   263→   263→        body: JSON.stringify({
   264→   264→          name: fullName,
   265→   265→          email: form.email,
   266→   266→          phone: form.phone,
   267→   267→          service: form.service,
   268→   268→          budget: form.budget,
   269→   269→          timeline: form.timeline,
   270→   270→          message,
   271→   271→          source,
   272→   272→          website: honeypot,
   273→   273→        }),
   274→   274→      });
   275→   275→      const payload = (await response.json()) as { ok?: boolean; id?: string; error?: string };
   276→   276→      if (!response.ok || !payload.ok) {
   277→   277→        throw new Error(payload.error ?? 'Something went wrong.');
   278→   278→      }
   279→   279→      trackEvent('generate_lead', {
   280→   280→        source,
   281→   281→        service: form.service || 'unspecified',
   282→   282→        budget: form.budget || 'unspecified',
   283→   283→        timeline: form.timeline || 'unspecified',
   284→   284→      });
   285→   285→      setLeadId(payload.id ?? null);
   286→   286→      setSubmitted(true);
   287→   287→      setForm({ ...EMPTY_FORM, service: defaultService ?? '' });
   288→   288→      toast({
   289→   289→        title: 'Request received!',
   290→   290→        description: 'Thanks — we will send your fixed quote within one business day.',
   291→   291→      });
   292→   292→    } catch (error) {
   293→   293→      toast({
   294→   294→        title: 'Could not send your request',
   295→   295→        description: error instanceof Error ? error.message : 'Please try again or email us directly.',
   296→   296→        variant: 'destructive',
   297→   297→      });
   298→   298→    } finally {
   299→   299→      setSubmitting(false);
   300→   300→    }
   301→   301→  };
   302→   302→
   303→   303→  const resetToForm = () => {
   304→   304→    setSubmitted(false);
   305→   305→    setLeadId(null);
   306→   306→    setErrors({});
   307→   307→  };
   308→   308→
   309→   309→  /* ---------------------------- styles ---------------------------- */
   310→   310→
   311→   311→  // The split layout pairs a dark rail with a white form (pricing / service
   312→   312→  // pages, full content width). `bare` is the clear single-column form —
   313→   313→  // the parent (contact page gradient-border card) supplies the surface.
   314→   314→  const wrapperClass = bare
   315→   315→    ? 'rounded-[24px]'
   316→   316→    : cn('card-soft grid overflow-hidden rounded-[24px] lg:grid-cols-[0.85fr_1.15fr]');
   317→   317→  const inputClass = dark
   318→   318→    ? 'h-11 rounded-xl border-white/15 bg-white/5 text-white placeholder:text-white/40 focus-visible:border-purple-400 focus-visible:ring-purple-400'
   319→   319→    : 'h-11 rounded-xl border-gray-200 bg-white text-[#0a0a0a] placeholder:text-gray-400 focus-visible:border-purple-500 focus-visible:ring-purple-500';
   320→   320→  const triggerClass = cn(inputClass, 'w-full data-[size=default]:h-11');
   321→   321→
   322→   322→  /* ------------------------- form fields -------------------------- */
   323→   323→
   324→   324→  const timelineField = (
   325→   325→    <Field id={`lead-timeline-${source}`} label="Timeline" optional>
   326→   326→      <Select value={form.timeline} onValueChange={set('timeline')}>
   327→   327→        <SelectTrigger
   328→   328→          id={`lead-timeline-${source}`}
   329→   329→          className={triggerClass}
   330→   330→          aria-label="Project timeline"
   331→   331→        >
   332→   332→          <SelectValue placeholder="When do you want to start?" />
   333→   333→        </SelectTrigger>
   334→   334→        <SelectContent>
   335→   335→          {TIMELINE_OPTIONS.map((option) => (
   336→   336→            <SelectItem key={option} value={option}>
   337→   337→              {option}
   338→   338→            </SelectItem>
   339→   339→          ))}
   340→   340→        </SelectContent>
   341→   341→      </Select>
   342→   342→    </Field>
   343→   343→  );
   344→   344→
   345→   345→  const companyField = (
   346→   346→    <Field id={`lead-company-${source}`} label="Company" optional>
   347→   347→      <Input
   348→   348→        id={`lead-company-${source}`}
   349→   349→        value={form.company}
   350→   350→        onChange={(e) => set('company')(e.target.value)}
   351→   351→        placeholder="Company or brand name"
   352→   352→        autoComplete="organization"
   353→   353→        className={inputClass}
   354→   354→      />
   355→   355→    </Field>
   356→   356→  );
   357→   357→
   358→   358→  const nameFields = (
   359→   359→    <>
   360→   360→      <Field
   361→   361→        id={`lead-first-${source}`}
   362→   362→        label="First name"
   363→   363→        required
   364→   364→        error={errors.name}
   365→   365→      >
   366→   366→        <Input
   367→   367→          id={`lead-first-${source}`}
   368→   368→          value={form.firstName}
   369→   369→          onChange={(e) => set('firstName')(e.target.value)}
   370→   370→          placeholder="Jane"
   371→   371→          autoComplete="given-name"
   372→   372→          required
   373→   373→          aria-invalid={Boolean(errors.name)}
   374→   374→          aria-describedby={errors.name ? `lead-first-${source}-error` : undefined}
   375→   375→          className={inputClass}
   376→   376→        />
   377→   377→      </Field>
   378→   378→      <Field id={`lead-last-${source}`} label="Last name" optional>
   379→   379→        <Input
   380→   380→          id={`lead-last-${source}`}
   381→   381→          value={form.lastName}
   382→   382→          onChange={(e) => set('lastName')(e.target.value)}
   383→   383→          placeholder="Smith"
   384→   384→          autoComplete="family-name"
   385→   385→          className={inputClass}
   386→   386→        />
   387→   387→      </Field>
   388→   388→    </>
   389→   389→  );
   390→   390→
   391→   391→  const contactFields = (
   392→   392→    <>
   393→   393→      <Field id={`lead-email-${source}`} label="Email" required error={errors.email}>
   394→   394→        <Input
   395→   395→          id={`lead-email-${source}`}
   396→   396→          type="email"
   397→   397→          value={form.email}
   398→   398→          onChange={(e) => set('email')(e.target.value)}
   399→   399→          placeholder="jane@company.com"
   400→   400→          autoComplete="email"
   401→   401→          required
   402→   402→          aria-invalid={Boolean(errors.email)}
   403→   403→          aria-describedby={errors.email ? `lead-email-${source}-error` : undefined}
   404→   404→          className={inputClass}
   405→   405→        />
   406→   406→      </Field>
   407→   407→      <Field id={`lead-phone-${source}`} label="Phone" optional>
   408→   408→        <Input
   409→   409→          id={`lead-phone-${source}`}
   410→   410→          type="tel"
   411→   411→          value={form.phone}
   412→   412→          onChange={(e) => set('phone')(e.target.value)}
   413→   413→          placeholder="+1 (555) 000-0000"
   414→   414→          autoComplete="tel"
   415→   415→          className={inputClass}
   416→   416→        />
   417→   417→      </Field>
   418→   418→    </>
   419→   419→  );
   420→   420→
   421→   421→  const pickFields = (
   422→   422→    <>
   423→   423→      <Field id={`lead-service-${source}`} label="Service" required error={errors.service}>
   424→   424→        <Select value={form.service} onValueChange={set('service')}>
   425→   425→          <SelectTrigger
   426→   426→            id={`lead-service-${source}`}
   427→   427→            className={triggerClass}
   428→   428→            aria-label="Service needed"
   429→   429→            aria-invalid={Boolean(errors.service)}
   430→   430→            aria-describedby={errors.service ? `lead-service-${source}-error` : undefined}
   431→   431→          >
   432→   432→            <SelectValue placeholder="What do you need?" />
   433→   433→          </SelectTrigger>
   434→   434→          <SelectContent>
   435→   435→            {services.map((service) => (
   436→   436→              <SelectItem key={service.slug} value={service.slug}>
   437→   437→                {service.name}
   438→   438→              </SelectItem>
   439→   439→            ))}
   440→   440→            <SelectItem value="not-sure">Not sure yet / something else</SelectItem>
   441→   441→          </SelectContent>
   442→   442→        </Select>
   443→   443→      </Field>
   444→   444→      <Field id={`lead-budget-${source}`} label="Budget" required error={errors.budget}>
   445→   445→        <Select value={form.budget} onValueChange={set('budget')}>
   446→   446→          <SelectTrigger
   447→   447→            id={`lead-budget-${source}`}
   448→   448→            className={triggerClass}
   449→   449→            aria-label="Budget range"
   450→   450→            aria-invalid={Boolean(errors.budget)}
   451→   451→            aria-describedby={errors.budget ? `lead-budget-${source}-error` : undefined}
   452→   452→          >
   453→   453→            <SelectValue placeholder="Select a range" />
   454→   454→          </SelectTrigger>
   455→   455→          <SelectContent>
   456→   456→            {BUDGET_RANGES.map((range) => (
   457→   457→              <SelectItem key={range} value={range}>
   458→   458→                {range}
   459→   459→              </SelectItem>
   460→   460→            ))}
   461→   461→          </SelectContent>
   462→   462→        </Select>
   463→   463→      </Field>
   464→   464→    </>
   465→   465→  );
   466→   466→
   467→   467→  const messageField = (
   468→   468→    <Field
   469→   469→      id={`lead-message-${source}`}
   470→   470→      label="Project details"
   471→   471→      required
   472→   472→      error={errors.message}
   473→   473→      className={compact ? undefined : 'sm:col-span-2'}
   474→   474→    >
   475→   475→      <Textarea
   476→   476→        id={`lead-message-${source}`}
   477→   477→        value={form.message}
   478→   478→        onChange={(e) => set('message')(e.target.value)}
   479→   479→        placeholder="Tell us about your project, goals, and timeline…"
   480→   480→        rows={compact ? 4 : 5}
   481→   481→        maxLength={MESSAGE_MAX}
   482→   482→        required
   483→   483→        aria-invalid={Boolean(errors.message)}
   484→   484→        aria-describedby={
   485→   485→          errors.message ? `lead-message-${source}-error` : `lead-message-${source}-hint`
   486→   486→        }
   487→   487→        className={cn(
   488→   488→          inputClass,
   489→   489→          'h-auto min-h-[120px] py-3 focus-visible:border-purple-500 focus-visible:ring-purple-500'
   490→   490→        )}
   491→   491→      />
