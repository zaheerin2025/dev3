    57→/** URL-safe slug: lowercase, alphanumerics and single dashes. */
    58→function slugify(value: string): string {
    59→  return value
    60→    .toLowerCase()
    61→    .replace(/[^a-z0-9]+/g, '-')
    62→    .replace(/(^-|-$)+/g, '');
    63→}

    65→/**
    66→ * Authorized fetch helper — the session rides in the HttpOnly cookie, so
    67→ * same-origin requests attach it automatically. Throws on non-OK, parses
    68→ * `{ ok, … }` JSON.
    69→ */
    70→async function adminFetch<T>(path: string, init?: RequestInit): Promise<T> {
    71→  const response = await fetch(path, {
    72→    ...init,
    73→    headers: { 'Content-Type': 'application/json', ...(init?.headers ?? {}) },
    74→  });
    75→  const payload = (await response.json().catch(() => null)) as
    76→    | ({ ok?: boolean; error?: string } & T)
    77→    | null;
    78→  if (!response.ok || !payload?.ok) {
    79→    throw new Error(payload?.error ?? 'Request failed.');
    80→  }
    81→  return payload;
    82→}

   107→/* ─────────────────────────── Login gate ─────────────────────────── */
   108→
   109→function AdminLogin({ onAuthenticated }: { onAuthenticated: () => void }) {
   110→  const { toast } = useToast();
   111→  const [passcode, setPasscode] = React.useState('');
   112→  const [loading, setLoading] = React.useState(false);
   113→
   114→  const submit = async (event: React.FormEvent) => {
   115→    event.preventDefault();
   116→    if (!passcode.trim() || loading) return;
   117→    setLoading(true);
   118→    try {
   119→      const response = await fetch('/api/admin/auth', {
   120→        method: 'POST',
   121→        headers: { 'Content-Type': 'application/json' },
   122→        body: JSON.stringify({ passcode: passcode.trim() }),
   123→      });
   124→      const payload = (await response.json().catch(() => null)) as {
   125→        ok?: boolean;
   126→        error?: string;
   127→      } | null;
   128→      if (!response.ok || !payload?.ok) {
   129→        throw new Error(payload?.error ?? 'Invalid passcode.');
   130→      }
   131→      onAuthenticated();
   132→    } catch (error) {
   133→      toast({
   134→        title: 'Sign in failed',
   135→        description: error instanceof Error ? error.message : 'Please try again.',
   136→        variant: 'destructive',
   137→      });
   138→    } finally {
   139→      setLoading(false);
   140→    }
   141→  };
   142→
   143→  return (
   144→    <div className="flex min-h-[60vh] items-center justify-center py-16">
   145→      <div className="card-surface w-full max-w-sm rounded-3xl p-8 shadow-[0_24px_48px_-24px_rgb(4_16_11/0.25)]">
   146→        <div className="flex flex-col items-center gap-1.5 text-center">
   147→          <img src="/logo.svg" alt="Developers3 logo" className="h-10 w-10" />
   148→          <h1 className="mt-2 text-xl font-bold tracking-tight">Admin panel</h1>
   149→          <p className="text-sm text-muted-foreground">
   150→            Sign in to manage posts, pricing, and site content.
   151→          </p>
   152→        </div>
   153→        <form onSubmit={submit} className="mt-6 flex flex-col gap-4">
   154→          <div className="flex flex-col gap-2">
   155→            <Label htmlFor="admin-passcode">Passcode</Label>
   156→            <Input
   157→              id="admin-passcode"
   158→              type="password"
   159→              value={passcode}
   160→              onChange={(event) => setPasscode(event.target.value)}
   161→              placeholder="Enter the admin passcode"
   162→              autoComplete="current-password"
   163→              autoFocus
   164→              required
   165→            />
   166→          </div>
   167→          <Button type="submit" className="h-11 w-full" disabled={loading}>
   168→            {loading ? (
   169→              <>
   170→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   171→                Signing in…
   172→              </>
   173→            ) : (
   174→              'Sign in'
   175→            )}
   176→          </Button>
   177→        </form>
   178→      </div>
   179→    </div>
   180→  );
   181→}

   617→        body: JSON.stringify({
   618→          entries: [
   619→            { key: 'pricing.websiteTiers', value: JSON.stringify(tiers) },
   620→            { key: 'pricing.servicePricingBlocks', value: JSON.stringify(blocks) },
   621→          ],
   622→        }),

   626→        title: 'Pricing updated — live on site',

   628→      });

   632→        title: 'Save failed',
   633→        description: error instanceof Error ? error.message : 'Please try again.',
   634→        variant: 'destructive',
   635→      });

   924→export function AdminView() {
   925→  const { toast } = useToast();
   926→  const [authed, setAuthed] = React.useState(false);
   927→  const [authChecked, setAuthChecked] = React.useState(false);
   928→

   930→  React.useEffect(() => {
   931→    let cancelled = false;
   932→    (async () => {
   933→      try {
   934→        const payload = await adminFetch<{ ok: boolean }>('/api/admin/auth');
   935→        if (!cancelled) setAuthed(Boolean(payload.ok));
   936→      } catch {
   937→        /* not signed in */
   938→      } finally {
   939→        if (!cancelled) setAuthChecked(true);

   943→      cancelled = true;
   944→    };

   949→      await fetch('/api/admin/auth', { method: 'DELETE' });

   951→      /* clearing client state regardless */
   952→    }
   953→    setAuthed(false);

   962→      variant: 'destructive',
