   194→   194→/**
   195→   195→ * Project inquiry form with honeypot spam protection.
   196→   196→ * Posts to /api/leads (stored in DB) and fires a GA4 generate_lead event.
   197→   197→ */
   198→   198→export function LeadForm({
   199→   199→  source = 'contact',
   200→   200→  defaultService,
   201→   201→  dark,
   202→   202→  compact,
   203→   203→  bare,
   204→   204→  className,
   205→   205→}: LeadFormProps) {
   206→   206→  const { toast } = useToast();
   207→   207→  const [form, setForm] = React.useState<FormState>(() => ({
   208→   208→    ...EMPTY_FORM,
   209→   209→    service: defaultService ?? '',
   210→   210→  }));
   211→   211→  const [honeypot, setHoneypot] = React.useState('');
   212→   212→  const [errors, setErrors] = React.useState<Errors>({});
   213→   213→  const [submitting, setSubmitting] = React.useState(false);
   214→   214→  const [submitted, setSubmitted] = React.useState(false);
   215→   215→  const [leadId, setLeadId] = React.useState<string | null>(null);
   216→   216→
   217→   217→  /** Map form fields to their validation error keys. */
   218→   218→  const ERROR_KEY_FOR: Partial<Record<keyof FormState, keyof Errors>> = {
   219→   219→    firstName: 'name',
   220→   220→    email: 'email',
   221→   221→    service: 'service',
   222→   222→    budget: 'budget',
   223→   223→    message: 'message',
   224→   224→  };
   225→   225→
   226→   226→  const set = (key: keyof FormState) => (value: string) => {
   227→   227→    setForm((prev) => ({ ...prev, [key]: value }));
   228→   228→    const errorKey = ERROR_KEY_FOR[key];
   229→   229→    if (!errorKey) return;
   230→   230→    setErrors((prev) => (prev[errorKey] ? { ...prev, [errorKey]: undefined } : prev));
   231→   231→  };
   232→   232→
   233→   233→  const validate = (): boolean => {
   234→   234→    const next: Errors = {};
   235→   235→    if (form.firstName.trim().length < 2) next.name = 'Please enter your first name.';
   236→   236→    if (!EMAIL_REGEX.test(form.email.trim())) next.email = 'Please enter a valid email address.';
   237→   237→    if (!form.service) next.service = 'Please select a service.';
   238→   238→    if (!form.budget) next.budget = 'Please select a budget range.';
   239→   239→    if (form.message.trim().length < MESSAGE_MIN) {
   240→   240→      next.message = `Please add a little more detail (at least ${MESSAGE_MIN} characters).`;
   241→   241→    }
   242→   242→    setErrors(next);
   243→   243→    return Object.keys(next).length === 0;
   244→   244→  };
   245→   245→
   246→   246→  const handleSubmit = async (event: React.FormEvent) => {
   247→   247→    event.preventDefault();
   248→   248→    if (!validate()) return;
   249→   249→    setSubmitting(true);
   250→   250→    try {
   251→   251→      const fullName = [form.firstName.trim(), form.lastName.trim()].filter(Boolean).join(' ');
   252→   252→      // Company is folded into the message so nothing the client shares gets lost.
   253→   253→      const message = form.company.trim()
   254→   254→        ? `Company: ${form.company.trim()}\n\n${form.message}`
   255→   255→        : form.message;
   256→   256→
   257→   257→      const response = await fetch('/api/leads', {
   258→   258→        method: 'POST',
   259→   259→        headers: { 'Content-Type': 'application/json' },
   260→   260→        body: JSON.stringify({
   261→   261→          name: fullName,
   262→   262→          email: form.email,
   263→   263→          phone: form.phone,
   264→   264→          service: form.service,
   265→   265→          budget: form.budget,
   266→   266→          timeline: form.timeline,
   267→   267→          message,
   268→   268→          source,
   269→   269→          website: honeypot,
   270→   270→        }),
   271→   271→      });
   272→   272→      const payload = (await response.json()) as { ok?: boolean; id?: string; error?: string };
   273→   273→      if (!response.ok || !payload.ok) {
   274→   274→        throw new Error(payload.error ?? 'Something went wrong.');
   275→   275→      }
   276→   276→      trackEvent('generate_lead', {
   277→   277→        source,
   278→   278→        service: form.service || 'unspecified',
   279→   279→        budget: form.budget || 'unspecified',
   280→   280→        timeline: form.timeline || 'unspecified',
   281→   281→      });
   282→   282→      setLeadId(payload.id ?? null);
   283→   283→      setSubmitted(true);
   284→   284→      setForm({ ...EMPTY_FORM, service: defaultService ?? '' });
   285→   285→      toast({
   286→   286→        title: 'Request received!',
   287→   287→        description: 'Thanks — we will send your fixed quote within one business day.',
   288→   288→      });
   289→   289→    } catch (error) {
   290→   290→      toast({
   291→   291→        title: 'Could not send your request',
   292→   292→        description: error instanceof Error ? error.message : 'Please try again or email us directly.',
   293→   293→        variant: 'destructive',
   294→   294→      });
   295→   295→    } finally {
   296→   296→      setSubmitting(false);
   297→   297→    }
   298→   298→  };
   299→   299→
   300→   300→  const resetToForm = () => {
   301→   301→    setSubmitted(false);
   302→   302→    setLeadId(null);
   303→   303→    setErrors({});
   304→   304→  };
   305→   305→
   306→   306→  /* ---------------------------- styles ---------------------------- */
   307→   307→
   308→   308→  // The split layout pairs a dark rail with a white form (pricing / service
   309→   309→  // pages, full content width). `bare` is the clear single-column form —
   310→   310→  // the parent (contact page gradient-border card) supplies the surface.
   311→   311→  const wrapperClass = bare
   312→   312→    ? 'rounded-[24px]'
   313→   313→    : cn('card-soft grid overflow-hidden rounded-[24px] lg:grid-cols-[0.85fr_1.15fr]');
   314→   314→  const inputClass = dark
   315→   315→    ? 'h-11 rounded-xl border-white/15 bg-white/5 text-white placeholder:text-white/40 focus-visible:border-purple-400 focus-visible:ring-purple-400'
   316→   316→    : 'h-11 rounded-xl border-gray-200 bg-white text-[#0a0a0a] placeholder:text-gray-400 focus-visible:border-purple-500 focus-visible:ring-purple-500';
   317→   317→  const triggerClass = cn(inputClass, 'w-full data-[size=default]:h-11');
   318→   318→
   319→   319→  /* ------------------------- form fields -------------------------- */
   320→   320→
   321→   321→  const timelineField = (
   322→   322→    <Field id={`lead-timeline-${source}`} label="Timeline" optional>
   323→   323→      <Select value={form.timeline} onValueChange={set('timeline')}>
   324→   324→        <SelectTrigger
   325→   325→          id={`lead-timeline-${source}`}
   326→   326→          className={triggerClass}
   327→   327→          aria-label="Project timeline"
   328→   328→        >
   329→   329→          <SelectValue placeholder="When do you want to start?" />
   330→   330→        </SelectTrigger>
   331→   331→        <SelectContent>
   332→   332→          {TIMELINE_OPTIONS.map((option) => (
   333→   333→            <SelectItem key={option} value={option}>
   334→   334→              {option}
   335→   335→            </SelectItem>
   336→   336→          ))}
   337→   337→        </SelectContent>
   338→   338→      </Select>
   339→   339→    </Field>
   340→   340→  );
   341→   341→
   342→   342→  const companyField = (
   343→   343→    <Field id={`lead-company-${source}`} label="Company" optional>
   344→   344→      <Input
   345→   345→        id={`lead-company-${source}`}
   346→   346→        value={form.company}
   347→   347→        onChange={(e) => set('company')(e.target.value)}
   348→   348→        placeholder="Company or brand name"
   349→   349→        autoComplete="organization"
   350→   350→        className={inputClass}
   351→   351→      />
   352→   352→    </Field>
   353→   353→  );
   354→   354→
   355→   355→  const nameFields = (
   356→   356→    <>
   357→   357→      <Field
   358→   358→        id={`lead-first-${source}`}
   359→   359→        label="First name"
   360→   360→        required
   361→   361→        error={errors.name}
   362→   362→      >
   363→   363→        <Input
   364→   364→          id={`lead-first-${source}`}
   365→   365→          value={form.firstName}
   366→   366→          onChange={(e) => set('firstName')(e.target.value)}
   367→   367→          placeholder="Jane"
   368→   368→          autoComplete="given-name"
   369→   369→          required
   370→   370→          aria-invalid={Boolean(errors.name)}
   371→   371→          aria-describedby={errors.name ? `lead-first-${source}-error` : undefined}
   372→   372→          className={inputClass}
   373→   373→        />
   374→   374→      </Field>
   375→   375→      <Field id={`lead-last-${source}`} label="Last name" optional>
   376→   376→        <Input
   377→   377→          id={`lead-last-${source}`}
   378→   378→          value={form.lastName}
   379→   379→          onChange={(e) => set('lastName')(e.target.value)}
   380→   380→          placeholder="Smith"
   381→   381→          autoComplete="family-name"
   382→   382→          className={inputClass}
   383→   383→        />
   384→   384→      </Field>
   385→   385→    </>
   386→   386→  );
   387→   387→
   388→   388→  const contactFields = (
   389→   389→    <>
   390→   390→      <Field id={`lead-email-${source}`} label="Email" required error={errors.email}>
   391→   391→        <Input
   392→   392→          id={`lead-email-${source}`}
   393→   393→          type="email"
   394→   394→          value={form.email}
   395→   395→          onChange={(e) => set('email')(e.target.value)}
   396→   396→          placeholder="jane@company.com"
   397→   397→          autoComplete="email"
   398→   398→          required
   399→   399→          aria-invalid={Boolean(errors.email)}
   400→   400→          aria-describedby={errors.email ? `lead-email-${source}-error` : undefined}
   401→   401→          className={inputClass}
   402→   402→        />
   403→   403→      </Field>
   404→   404→      <Field id={`lead-phone-${source}`} label="Phone" optional>
   405→   405→        <Input
   406→   406→          id={`lead-phone-${source}`}
   407→   407→          type="tel"
   408→   408→          value={form.phone}
   409→   409→          onChange={(e) => set('phone')(e.target.value)}
   410→   410→          placeholder="+1 (555) 000-0000"
   411→   411→          autoComplete="tel"
   412→   412→          className={inputClass}
   413→   413→        />
   414→   414→      </Field>
   415→   415→    </>
   416→   416→  );
   417→   417→
   418→   418→  const pickFields = (
   419→   419→    <>
   420→   420→      <Field id={`lead-service-${source}`} label="Service" required error={errors.service}>
   421→   421→        <Select value={form.service} onValueChange={set('service')}>
   422→   422→          <SelectTrigger
   423→   423→            id={`lead-service-${source}`}
   424→   424→            className={triggerClass}
   425→   425→            aria-label="Service needed"
   426→   426→            aria-invalid={Boolean(errors.service)}
   427→   427→            aria-describedby={errors.service ? `lead-service-${source}-error` : undefined}
   428→   428→          >
   429→   429→            <SelectValue placeholder="What do you need?" />
   430→   430→          </SelectTrigger>
   431→   431→          <SelectContent>
   432→   432→            {services.map((service) => (
   433→   433→              <SelectItem key={service.slug} value={service.slug}>
   434→   434→                {service.name}
   435→   435→              </SelectItem>
   436→   436→            ))}
   437→   437→            <SelectItem value="not-sure">Not sure yet / something else</SelectItem>
   438→   438→          </SelectContent>
   439→   439→        </Select>
   440→   440→      </Field>
   441→   441→      <Field id={`lead-budget-${source}`} label="Budget" required error={errors.budget}>
   442→   442→        <Select value={form.budget} onValueChange={set('budget')}>
   443→   443→          <SelectTrigger
   444→   444→            id={`lead-budget-${source}`}
   445→   445→            className={triggerClass}
   446→   446→            aria-label="Budget range"
   447→   447→            aria-invalid={Boolean(errors.budget)}
   448→   448→            aria-describedby={errors.budget ? `lead-budget-${source}-error` : undefined}
   449→   449→          >
   450→   450→            <SelectValue placeholder="Select a range" />
   451→   451→          </SelectTrigger>
   452→   452→          <SelectContent>
   453→   453→            {BUDGET_RANGES.map((range) => (
   454→   454→              <SelectItem key={range} value={range}>
   455→   455→                {range}
   456→   456→              </SelectItem>
   457→   457→            ))}
   458→   458→          </SelectContent>
   459→   459→        </Select>
   460→   460→      </Field>
   461→   461→    </>
   462→   462→  );
   463→   463→
   464→   464→  const messageField = (
   465→   465→    <Field
   466→   466→      id={`lead-message-${source}`}
   467→   467→      label="Project details"
   468→   468→      required
   469→   469→      error={errors.message}
   470→   470→      className={compact ? undefined : 'sm:col-span-2'}
   471→   471→    >
   472→   472→      <Textarea
   473→   473→        id={`lead-message-${source}`}
   474→   474→        value={form.message}
   475→   475→        onChange={(e) => set('message')(e.target.value)}
   476→   476→        placeholder="Tell us about your project, goals, and timeline…"
   477→   477→        rows={compact ? 4 : 5}
   478→   478→        maxLength={MESSAGE_MAX}
   479→   479→        required
   480→   480→        aria-invalid={Boolean(errors.message)}
   481→   481→        aria-describedby={
   482→   482→          errors.message ? `lead-message-${source}-error` : `lead-message-${source}-hint`
   483→   483→        }
   484→   484→        className={cn(
   485→   485→          inputClass,
   486→   486→          'h-auto min-h-[120px] py-3 focus-visible:border-purple-500 focus-visible:ring-purple-500'
   487→   487→        )}
   488→   488→      />
   489→   489→      <div
   490→   490→        id={`lead-message-${source}-hint`}
