    35→/** Deterministic heading → element id helper for the in-page table of contents. */
    36→function slugify(heading: string): string {
    37→  return heading
    38→    .toLowerCase()
    39→    .replace(/[^a-z0-9]+/g, '-')
    40→    .replace(/(^-|-$)/g, '');
    41→}
    42→
    43→/** Renders the last word of the title with the light gradient (dark hero). */
    44→function GradientLastWord({ title }: { title: string }) {
    45→  const splitAt = title.lastIndexOf(' ');
    46→  if (splitAt === -1) return <span className="text-gradient-soft">{title}</span>;
    47→  return (
    48→    <>
    49→      {title.slice(0, splitAt + 1)}
    50→      <span className="text-gradient-soft">{title.slice(splitAt + 1)}</span>
    51→    </>
    52→  );
    53→}

    61→  );
