    51→function minifyCss(css: string): string {
    52→  try {
    53→    const stash: string[] = [];
    54→    const keep = (raw: string) => {
    55→      stash.push(raw);
    56→      return '\u0000' + (stash.length - 1) + '\u0000';
    57→    };
    58→    let out = '';
    59→    let i = 0;
    60→    const n = css.length;
    61→    while (i < n) {
    62→      const c = css[i];
    63→      if (c === '"' || c === "'") {
    64→        let j = i + 1;
    65→        while (j < n && css[j] !== c) {
    66→          if (css[j] === '\\') j++;
    67→          j++;
    68→        }
    69→        out += keep(css.slice(i, Math.min(j + 1, n)));
    70→        i = j + 1;
    71→        continue;
    72→      }
    73→      if (c === '/' && css[i + 1] === '*') {
    74→        const end = css.indexOf('*/', i + 2);
    75→        out += ' ';
    76→        i = end === -1 ? n : end + 2;
    77→        continue;
    78→      }
    79→      if (css.slice(i, i + 5).toLowerCase() === 'calc(') {
    80→        let depth = 0;
    81→        let j = i + 4;
    82→        for (; j < n; j++) {
    83→          if (css[j] === '(') depth++;
    84→          else if (css[j] === ')') {
    85→            depth--;
    86→            if (depth === 0) break;
    87→          }
    88→        }
    89→        out += keep(css.slice(i, Math.min(j + 1, n)));
    90→        i = j + 1;
    91→        continue;
    92→      }
    93→      out += c;
    94→      i++;
    95→    }
    96→    out = out.replace(/\s+/g, ' ').trim();
    97→    out = out.replace(/\s*([{}:;,>+~])\s*/g, '$1');
    98→    out = out.replace(/;}/g, '}');
    99→    out = out.replace(/\u0000(\d+)\u0000/g, (_, d: string) => stash[Number(d)]);
   100→    return out;
   101→  } catch (e) {
   102→    return '⚠ Minify failed: ' + (e as Error).message;
   103→  }
   104→}

   137→function regexAllowedAfter(out: string): boolean {
   138→  let i = out.length - 1;
   139→  while (i >= 0 && /\s/.test(out[i])) i--;
   140→  if (i < 0) return true;
   141→  const c = out[i];
   142→  if ('.,=:([!&|?{};+-*%<>~^'.includes(c)) return true;
   143→  let j = i;
   144→  while (j >= 0 && /[A-Za-z0-9_$]/.test(out[j])) j--;
   145→  const word = out.slice(j + 1, i + 1);
   146→  return word !== '' && JS_KEYWORDS_BEFORE_REGEX.has(word);
   147→}

   149→function stripJsComments(src: string): string {
   150→  let out = '';
   151→  let i = 0;
   152→  const n = src.length;
   153→  while (i < n) {
   154→    const c = src[i];
   155→    if (c === '"' || c === "'") {
   156→      let j = i + 1;
   157→      while (j < n && src[j] !== c) {
   158→        if (src[j] === '\\') j++;
   159→        j++;
   160→      }
   161→      out += src.slice(i, Math.min(j + 1, n));
   162→      i = j + 1;
   163→      continue;
   164→    }
   165→    if (c === '`') {
   166→      let j = i + 1;
   167→      while (j < n && src[j] !== '`') {
   168→        if (src[j] === '\\') j++;
   169→        j++;
   170→      }
   171→      out += src.slice(i, Math.min(j + 1, n));
   172→      i = j + 1;
   173→      continue;
   174→    }
   175→    if (c === '/' && src[i + 1] === '/') {
   176→      const nl = src.indexOf('\n', i);
   177→      i = nl === -1 ? n : nl; // keep the newline itself
   178→      continue;
   179→    }
   180→    if (c === '/' && src[i + 1] === '*') {
   181→      const end = src.indexOf('*/', i + 2);
   182→      out += ' ';
   183→      i = end === -1 ? n : end + 2;
   184→      continue;
   185→    }
   186→    if (c === '/' && regexAllowedAfter(out)) {
   187→      let j = i + 1;
   188→      let inClass = false;
   189→      while (j < n) {
   190→        const ch = src[j];
   191→        if (ch === '\\') {
   192→          j += 2;
   193→          continue;
   194→        }
   195→        if (ch === '[') inClass = true;
   196→        else if (ch === ']') inClass = false;
   197→        else if (ch === '/' && !inClass) break;
   198→        j++;
   199→      }
   200→      let k = j + 1;
   201→      while (k < n && /[a-z]/i.test(src[k])) k++; // flags
   202→      out += src.slice(i, Math.min(k, n));
   203→      i = k;
   204→      continue;
   205→    }
   206→    out += c;
   207→    i++;
   208→  }
   209→  return out;
   210→}

   212→function minifyJsSafe(src: string): string {
   213→  try {
   214→    const stripped = stripJsComments(src);
   215→    const lines = stripped.split('\n').map((l) => l.replace(/[ \t]+$/, ''));
   216→    const out: string[] = [];
   217→    let blanks = 0;
   218→    for (const l of lines) {
   219→      if (l.trim() === '') {
   220→        blanks++;
   221→        if (blanks === 1 && out.length > 0) out.push('');
   222→      } else {
   223→        blanks = 0;
   224→        out.push(l);
   225→      }
   226→    }
   227→    while (out.length > 0 && out[out.length - 1] === '') out.pop();
   228→    return out.join('\n');
   229→  } catch (e) {
   230→    return '⚠ Minify failed: ' + (e as Error).message;
   231→  }
   232→}

   263→const HTML_VOID = new Set([
   264→  'area', 'base', 'br', 'col', 'embed', 'hr', 'img', 'input', 'link', 'meta', 'param', 'source', 'track', 'wbr',
   265→]);
   266→const HTML_INLINE = new Set([
   267→  'a', 'span', 'em', 'strong', 'b', 'i', 'u', 's', 'small', 'sub', 'sup', 'code', 'kbd', 'q',
   268→  'abbr', 'cite', 'mark', 'time', 'td', 'th', 'option', 'label', 'del', 'ins', 'bdi', 'bdo',
   269→]);
   270→const RAW_ELEMENTS = new Set(['script', 'style', 'pre']);
   271→
   272→function tokenizeHtml(src: string): string[] {
   273→  const tokens: string[] = [];
   274→  let i = 0;
   275→  const n = src.length;
   276→  while (i < n) {
   277→    if (src.startsWith('<!--', i)) {
   278→      const end = src.indexOf('-->', i + 4);
   279→      const stop = end === -1 ? n : end + 3;
   280→      tokens.push(src.slice(i, stop));
   281→      i = stop;
   282→      continue;
   283→    }
   284→    if (src[i] === '<') {
   285→      const m = /^<\/?([a-zA-Z][a-zA-Z0-9-]*)/.exec(src.slice(i, i + 40));
   286→      if (m) {
   287→        const name = m[1].toLowerCase();
   288→        if (RAW_ELEMENTS.has(name) && !src.startsWith('</', i)) {
   289→          const close = src.toLowerCase().indexOf('</' + name, i + 1);
   290→          const stop = close === -1 ? n : close + name.length + 3;
   291→          tokens.push(src.slice(i, stop));
   292→          i = stop;
   293→          continue;
   294→        }
   295→        const end = src.indexOf('>', i);
   296→        const stop = end === -1 ? n : end + 1;
   297→        tokens.push(src.slice(i, stop));
   298→        i = stop;
   299→        continue;
   300→      }
   301→    }
   302→    const next = src.indexOf('<', i + 1);
   303→    const stop = next === -1 ? n : next;
   304→    if (stop > i) tokens.push(src.slice(i, stop));
   305→    i = stop > i ? stop : i + 1;
   306→  }
   307→  return tokens;
   308→}

   310→function formatHtml(src: string): string {
   311→  try {
   312→    const tokens = tokenizeHtml(src);
   313→    const out: string[] = [];
   314→    let depth = 0;
   315→    let line: string | null = null;
   316→    const flush = () => {
   317→      if (line !== null && line.trim() !== '') {
   318→        out.push('  '.repeat(Math.max(0, depth)) + line.trim());
   319→      }
   320→      line = null;
   321→    };
   322→    for (const tok of tokens) {
   323→      if (tok.startsWith('<!--')) {
   324→        flush();
   325→        out.push('  '.repeat(depth) + tok.trim());
   326→        continue;
   327→      }
   328→      if (tok.startsWith('<')) {
   329→        const m = /^<\/?([a-zA-Z][a-zA-Z0-9-]*)/.exec(tok.slice(0, 40));
   330→        const name = (m?.[1] ?? '').toLowerCase();
   331→        const closing = tok.startsWith('</');
   332→        const selfClosing = /\/>$/.test(tok) || HTML_VOID.has(name);
   333→        if (RAW_ELEMENTS.has(name) && !closing) {
   334→          flush();
   335→          const lines = tok.split('\n').map((l) => l.replace(/[ \t]+$/, ''));
   336→          out.push('  '.repeat(depth) + lines[0]);
   337→          for (let k = 1; k < lines.length - 1; k++) out.push(lines[k]);
   338→          if (lines.length > 1) out.push('  '.repeat(depth) + lines[lines.length - 1]);
   339→          continue;
   340→        }
   341→        if (HTML_INLINE.has(name) || (selfClosing && (name === 'br' || name === 'wbr' || name === 'img'))) {
   342→          line = (line ?? '') + tok.replace(/\s+/g, ' ');
   343→          continue;
   344→        }
   345→        flush();
   346→        const tag = tok.replace(/\s+/g, ' ').trim();
   347→        if (closing) {
   348→          depth = Math.max(0, depth - 1);
   349→          out.push('  '.repeat(depth) + tag);
   350→        } else if (selfClosing) {
   351→          out.push('  '.repeat(depth) + tag);
   352→        } else {
   353→          out.push('  '.repeat(depth) + tag);
   354→          depth++;
   355→        }
   356→      } else {
   357→        const text = tok.replace(/\s+/g, ' ').trim();
   358→        if (text !== '') line = (line ?? '') + text;
   359→      }
   360→    }
   361→    flush();
   362→    const cleaned: string[] = [];
   363→    for (const l of out) {
   364→      if (l === '' && cleaned[cleaned.length - 1] === '') continue;
   365→      cleaned.push(l);
   366→    }
   367→    while (cleaned.length > 0 && cleaned[cleaned.length - 1] === '') cleaned.pop();
   368→    return cleaned.join('\n');
   369→  } catch (e) {
   370→    return '⚠ Format failed: ' + (e as Error).message;
   371→  }
   372→}

   399→function base64Encode(text: string): string {
   400→  const bytes = new TextEncoder().encode(text);
   401→  let bin = '';
   402→  const CHUNK = 0x8000;
   403→  for (let i = 0; i < bytes.length; i += CHUNK) {
   404→    bin += String.fromCharCode(...bytes.subarray(i, i + CHUNK));
   405→  }
   406→  return btoa(bin);
   407→}
   408→
   409→function base64Decode(text: string): string {
   410→  let clean = text.trim().replace(/\s+/g, '');
   411→  const dataMatch = /^data:[^,]*,/i.exec(clean);
   412→  if (dataMatch) clean = clean.slice(dataMatch[0].length);
   413→  const bin = atob(clean);
   414→  const bytes = Uint8Array.from(bin, (ch) => ch.charCodeAt(0));
   415→  return new TextDecoder().decode(bytes);
   416→}

   463→function codeAlpha(upperStart: number, lowerStart: number, digitStart: number | null, exceptions: Record<string, string> = {}) {
   464→  return (ch: string): string => {
   465→    if (exceptions[ch] !== undefined) return exceptions[ch];
   466→    const c = ch.codePointAt(0)!;
   467→    if (c >= 65 && c <= 90) return String.fromCodePoint(upperStart + (c - 65));
   468→    if (c >= 97 && c <= 122) return String.fromCodePoint(lowerStart + (c - 97));
   469→    if (digitStart !== null && c >= 48 && c <= 57) return String.fromCodePoint(digitStart + (c - 48));
   470→    return ch;
   471→  };
   472→}
   473→
   474→const CIRCLED_MAP = (ch: string): string => {
   475→  const c = ch.codePointAt(0)!;
   476→  if (c >= 65 && c <= 90) return String.fromCodePoint(0x24b6 + (c - 65));
   477→  if (c >= 97 && c <= 122) return String.fromCodePoint(0x24d0 + (c - 97));
   478→  if (c === 48) return '⓪';
   479→  if (c >= 49 && c <= 57) return String.fromCodePoint(0x2460 + (c - 49));
   480→  return ch;
   481→};

   805→const TITLE_MINOR = new Set(['a', 'an', 'and', 'as', 'at', 'but', 'by', 'for', 'in', 'nor', 'of', 'on', 'or', 'per', 'the', 'to', 'vs', 'via']);

   807→function caseTokens(s: string): string[] {
   808→  return s
   809→    .replace(/([A-Z]+)([A-Z][a-z])/g, '$1 $2')
   810→    .replace(/([a-z0-9])([A-Z])/g, '$1 $2')
   811→    .split(/[^A-Za-z0-9]+/)
   812→    .filter(Boolean);
   813→}
   814→
   815→function convertCases(input: string): string {
   816→  const toks = caseTokens(input);
   817→  const lower = toks.map((w) => w.toLowerCase());
   818→  const camel = lower.map((w, i) => (i === 0 ? w : w.charAt(0).toUpperCase() + w.slice(1))).join('');
   819→  const pascal = lower.map((w) => w.charAt(0).toUpperCase() + w.slice(1)).join('');
   820→  const snake = lower.join('_');
   821→  const kebab = lower.join('-');
   822→  const constant = lower.join('_').toUpperCase();
   823→  const title = toks
   824→    .map((w, i) => (i > 0 && TITLE_MINOR.has(w.toLowerCase()) ? w.toLowerCase() : w.charAt(0).toUpperCase() + w.slice(1).toLowerCase()))
   825→    .join(' ');
   826→  const sentence = input.toLowerCase().replace(/(^\s*[a-z])|([.!?]\s+[a-z])/g, (m) => m.toUpperCase());
   827→  const upper = input.toUpperCase();
   828→  const lowerAll = input.toLowerCase();
   829→  let altIdx = 0;
   830→  const alternating = input
   831→    .split('')
   832→    .map((ch) => (/[a-zA-Z]/.test(ch) ? (altIdx++ % 2 === 0 ? ch.toLowerCase() : ch.toUpperCase()) : ch))
   833→    .join('');
   834→  return [
   835→    'camelCase: ' + (camel || input),
   836→    'PascalCase: ' + (pascal || input),
   837→    'snake_case: ' + (snake || input),
   838→    'kebab-case: ' + (kebab || input),
   839→    'CONSTANT_CASE: ' + (constant || input),
   840→    'Title Case: ' + (title || input),
   841→    'Sentence case: ' + (sentence || input),
   842→    'UPPER CASE: ' + upper,
   843→    'lower case: ' + lowerAll,
   844→    'aLtErNaTiNg: ' + alternating,
   845→  ].join('\n');
   846→}

   865→function splitTopLevel(s: string): string[] {
   866→  const parts: string[] = [];
   867→  let depth = 0;
   868→  let cur = '';
   869→  for (const ch of s) {
   870→    if (ch === '(') {
   871→      depth++;
   872→      cur += ch;
   873→    } else if (ch === ')') {
   874→      depth--;
   875→      cur += ch;
   876→    } else if (ch === ',' && depth === 0) {
   877→      parts.push(cur);
   878→      cur = '';
   879→    } else cur += ch;
   880→  }
   881→  if (cur.trim() !== '') parts.push(cur);
   882→  return parts;
   883→}

   885→function formatSql(input: string): string {
   886→  try {
   887→    const stash: string[] = [];
   888→    let flat = input.replace(/'(?:[^']|'')*'/g, (m) => {
   889→      stash.push(m);
   890→      return '\u0001' + (stash.length - 1) + '\u0001';
   891→    });
   892→    flat = flat.replace(/\s+/g, ' ').trim();
   893→    const KEYWORDS = [
   894→      'SELECT', 'INSERT INTO', 'DELETE', 'UPDATE', 'SET', 'VALUES', 'FROM', 'WHERE', 'GROUP BY',
   895→      'ORDER BY', 'HAVING', 'LIMIT', 'OFFSET', 'INNER JOIN', 'LEFT OUTER JOIN', 'RIGHT OUTER JOIN',
   896→      'FULL OUTER JOIN', 'LEFT JOIN', 'RIGHT JOIN', 'FULL JOIN', 'CROSS JOIN', 'OUTER JOIN', 'JOIN',
   897→      'ON', 'AND', 'OR', 'NOT', 'IN', 'LIKE', 'BETWEEN', 'IS', 'NULL', 'AS', 'ASC', 'DESC',
   898→      'UNION ALL', 'UNION', 'DISTINCT', 'CASE', 'WHEN', 'THEN', 'ELSE', 'END', 'EXISTS', 'BY',
   899→    ];
   900→    for (const kw of KEYWORDS) {
   901→      flat = flat.replace(new RegExp('\\b' + kw + '\\b', 'g'), kw);
   902→    }
   903→    // split the (first) SELECT list onto indented lines
   904→    const upperFlat = flat.toUpperCase();
   905→    const selIdx = upperFlat.indexOf('SELECT');
   906→    if (selIdx >= 0) {
   907→      let depth = 0;
   908→      let i = selIdx + 6;
   909→      let fromIdx = -1;
   910→      while (i < flat.length) {
   911→        const ch = flat[i];
   912→        if (ch === '(') depth++;
   913→        else if (ch === ')') depth--;
   914→        else if (depth === 0 && flat.slice(i, i + 5).toUpperCase() === ' FROM') {
   915→          fromIdx = i;
   916→          break;
   917→        }
   918→        i++;
   919→      }
   920→      if (fromIdx > selIdx) {
   921→        const cols = flat.slice(selIdx + 6, fromIdx).trim();
   922→        const splitCols = splitTopLevel(cols)
   923→          .map((c) => c.trim())
   924→          .filter((c) => c !== '')
   925→          .join(',\n    ');
   926→        flat = flat.slice(0, selIdx + 6) + ' ' + splitCols + ' ' + flat.slice(fromIdx + 1);
   927→      }
   928→    }
   929→    flat = flat.replace(
   930→      /\b(GROUP BY|ORDER BY|INNER JOIN|LEFT OUTER JOIN|RIGHT OUTER JOIN|FULL OUTER JOIN|LEFT JOIN|RIGHT JOIN|FULL JOIN|CROSS JOIN|OUTER JOIN|FROM|WHERE|HAVING|LIMIT|OFFSET|UNION ALL|UNION|VALUES|SET|JOIN)\b/g,
   931→      '\n$1'
   932→    );
   933→    flat = flat.replace(/\bON\b/g, '\n  ON');
   934→    flat = flat.replace(/\b(AND|OR)\b/g, '\n  $1');
   935→    let lines = flat.split('\n').map((l) => l.replace(/[ \t]+$/g, ''));
   936→    lines = lines.filter((l, idx) => l.trim() !== '' || idx === 0);
   937→    let result = lines.join('\n');
   938→    result = result.replace(/\u0001(\d+)\u0001/g, (_, d: string) => stash[Number(d)]);
   939→    return result;
   940→  } catch (e) {
   941→    return '⚠ Format failed: ' + (e as Error).message;
   942→  }
   943→}

  1248→function weightedLength(text: string): number {
  1249→  let total = 0;
  1250→  const ranges: [number, number][] = [];

  1252→  let m: RegExpExecArray | null;
  1253→  while ((m = re.exec(text)) !== null) ranges.push([m.index, m.index + m[0].length]);
  1254→  let i = 0;
  1255→  const advance = (stop: number) => {
  1256→    while (i < stop) {
  1257→      const cp = text.codePointAt(i)!;
  1258→      total += cp > 0x2000 ? 2 : 1;
  1259→      i += cp > 0xffff ? 2 : 1;

  1262→  for (const [s, e] of ranges) {
  1263→    advance(s);
  1264→    total += 23;

  1271→function splitThread(text: string, limit = 275): string[] {

  1280→    let buf = '';

  1282→      if (buf !== '') chunks.push(buf);
