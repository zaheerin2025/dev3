    58→/**
    59→ * Splits the H1 so the service keyword phrase carries the brand gradient.
    60→ * Tries the full service name, then progressively shorter prefixes (min. two
    61→ * words) so phrases like "Website Maintenance & Support" still match their
    62→ * mention inside the hero title. Falls back to the first up-to-3 words.
    63→ * The returned tuple keeps the title text byte-for-byte intact.
    64→ */
    65→function splitHeroTitle(title: string, phrase: string): [string, string, string] {
    66→  const lower = title.toLowerCase();
    67→  const phraseWords = phrase.split(' ');
    68→  for (let len = Math.min(phraseWords.length, 4); len >= 2; len -= 1) {
    69→    const candidate = phraseWords.slice(0, len).join(' ').toLowerCase();
    70→    const idx = lower.indexOf(candidate);
    71→    if (idx === -1) continue;
    72→    const end = idx + candidate.length;
    73→    const boundaryBefore = idx === 0 || lower[idx - 1] === ' ';
    74→    const boundaryAfter = end === lower.length || lower[end] === ' ';
    75→    if (boundaryBefore && boundaryAfter) {
    76→      return [title.slice(0, idx), title.slice(idx, end), title.slice(end)];
    77→    }
    78→  }
    79→  const words = title.split(' ');
    80→  const head = words.slice(0, Math.min(3, words.length)).join(' ');
    81→  const tail = words.length > 3 ? ` ${words.slice(3).join(' ')}` : '';
    82→  return ['', head, tail];
    83→}
