   348→        </p>
   349→        <p className={cn('leading-relaxed', compact ? 'text-xs text-white/60' : 'text-sm text-white/60')}>
   350→          Pick a service to see a live estimate — or tell us what you need and we&rsquo;ll scope it
   351→          with you on a free call.
   352→        </p>
   353→      </div>
   354→    );
   355→  }
   356→
   357→  return (
   358→    <div className="flex flex-col gap-2">
   359→      <p className={cn('text-gradient font-display font-extrabold tracking-tight', compact ? 'text-2xl sm:text-3xl' : 'text-4xl')}>
   360→        ${estimate.low.toLocaleString('en-US')} – ${estimate.high.toLocaleString('en-US')}
   361→        <span className="text-lg font-bold">{estimate.suffix}</span>
   362→      </p>
   363→      <p className={cn('font-semibold uppercase tracking-[0.14em]', compact ? 'text-[10px] text-white/50' : 'text-xs text-white/50')}>
   364→        {estimate.caption} · guidance range
   365→      </p>
   366→      {!compact ? (
   367→        <ul className="mt-3 flex flex-col gap-1.5">
   368→          {estimate.lines.map((line) => (
   369→            <li key={line.label} className="flex items-center justify-between gap-3 text-[13px]">
   370→              <span className="min-w-0 truncate text-white/60">{line.label}</span>
   371→              <span
   372→                className={cn(
   373→                  'shrink-0 font-mono font-semibold tabular-nums',
   374→                  line.tone === 'base' && 'text-white',
   375→                  line.tone === 'up' && 'text-emerald-300',
   376→                  line.tone === 'down' && 'text-emerald-300',
   377→                  line.tone === 'neutral' && 'text-white/80'
   378→                )}
   379→              >
   380→                {line.value}
   381→              </span>
   382→            </li>
   383→          ))}
   384→        </ul>
   385→      ) : null}
   386→    </div>
   387→  );
   388→}

   394→/**
   395→ * The global instant-quote experience.
   396→ * - DYNAMIC: every service drives its own project types, scope sizes and
   397→ *   add-ons from src/data/quote-config.ts — the form reshapes per service.
   398→ * - AUTOMATED: a live estimate range recalculates on every selection.
   399→ * - COHESIVE: one component renders the quote section on every service page,
   400→ *   the pricing page, and the contact page (inline variant).
   401→ */
   402→export function QuoteBuilder({
   403→  source = 'contact',
   404→  defaultService,
   405→  variant = 'full',
   406→  className,
   407→}: QuoteBuilderProps) {
   408→  const { toast } = useToast();
   409→  const inline = variant === 'inline';
   410→  const idPrefix = `qb-${source.replace(/[^a-z0-9]+/gi, '-')}`;
   411→
   412→  const [serviceSlug, setServiceSlug] = React.useState(defaultService ?? NOT_SURE_SLUG);
   413→  const [selection, setSelection] = React.useState(() => defaultsFor(getQuoteConfig(defaultService ?? NOT_SURE_SLUG)));
   414→  const [pickedAddons, setPickedAddons] = React.useState<string[]>([]);
   415→  const [genericBudget, setGenericBudget] = React.useState('');
   416→  const [genericTimeline, setGenericTimeline] = React.useState('');
   417→  const [form, setForm] = React.useState<FormState>(EMPTY_FORM);
   418→  const [honeypot, setHoneypot] = React.useState('');
   419→  const [errors, setErrors] = React.useState<Errors>({});
   420→  const [submitting, setSubmitting] = React.useState(false);
   421→  const [submitted, setSubmitted] = React.useState(false);
   422→  const [leadId, setLeadId] = React.useState<string | null>(null);
   423→
   424→  const config = getQuoteConfig(serviceSlug);
   425→  const paceOptions = getPaceOptions(config);
   426→  const selectedService = services.find((service) => service.slug === serviceSlug);
   427→  const serviceName = selectedService?.name ?? 'Not sure yet / something else';
   428→  const estimate = config
   429→    ? computeEstimate(config, { ...selection, addons: pickedAddons })
   430→    : null;
   431→
   432→  /* ------------------------- interactions ------------------------- */
   433→
   434→  const pickService = (slug: string) => {
   435→    setServiceSlug(slug);
   436→    setSelection(defaultsFor(getQuoteConfig(slug)));
   437→    setPickedAddons([]);
   438→    setGenericBudget('');
   439→    setGenericTimeline('');
   440→    setErrors({});
   441→  };
   442→
   443→  const setFormValue = (key: keyof FormState) => (value: string) => {
   444→    setForm((prev) => ({ ...prev, [key]: value }));
   445→    const errorKey = key === 'firstName' ? 'name' : key === 'notes' ? 'notes' : key === 'email' ? 'email' : null;
   446→    if (errorKey) {
   447→      setErrors((prev) => (prev[errorKey] ? { ...prev, [errorKey]: undefined } : prev));
   448→    }
   449→  };
   450→
   451→  const validate = (): boolean => {
   452→    const next: Errors = {};
   453→    if (form.firstName.trim().length < 2) next.name = 'Please enter your first name.';
   454→    if (!EMAIL_REGEX.test(form.email.trim())) next.email = 'Please enter a valid email address.';
   455→    if (!config && !genericBudget) next.budget = 'Please pick a budget range.';
   456→    if ((!config || !estimate) && form.notes.trim().length < NOTES_MIN_WHEN_REQUIRED) {
   457→      next.notes = 'Please add a few details about what you need (at least 10 characters).';
   458→    }
   459→    setErrors(next);
   460→    return Object.keys(next).length === 0;
   461→  };
   462→
   463→  const handleSubmit = async (event: React.FormEvent) => {
   464→    event.preventDefault();
   465→    if (!validate()) return;
   466→    setSubmitting(true);
   467→    try {
   468→      const fullName = [form.firstName.trim(), form.lastName.trim()].filter(Boolean).join(' ');
   469→
   470→      const summaryLines = describeSelection(serviceName, config, { ...selection, addons: pickedAddons }, estimate);
   471→      const message = [
   472→        form.company.trim() ? `Company: ${form.company.trim()}` : null,
   473→        '',
   474→        '— Quote configuration —',
   475→        ...summaryLines,
   476→        '',
   477→        form.notes.trim() ? `Notes: ${form.notes.trim()}` : 'Notes: (none added)',
   478→      ]
   479→        .filter((line) => line !== null)
   480→        .join('\n');
   481→
   482→      const response = await fetch('/api/leads', {
   483→        method: 'POST',
   484→        headers: { 'Content-Type': 'application/json' },
   485→        body: JSON.stringify({
   486→          name: fullName,
   487→          email: form.email,
   488→          phone: form.phone,
   489→          service: serviceSlug === NOT_SURE_SLUG ? 'not-sure' : serviceSlug,
   490→          budget: estimate ? estimateRangeLabel(estimate) : genericBudget,
   491→          timeline: config ? paceOptions.find((p) => p.value === selection.pace)?.label ?? '' : genericTimeline,
   492→          message,
   493→          estimate: estimate ? `${estimateRangeLabel(estimate)} ${estimate.caption}` : '',
   494→          quoteConfig: serializeSelection(serviceName, config, { ...selection, addons: pickedAddons }, estimate),
   495→          source,
   496→          website: honeypot,
   497→        }),
   498→      });
   499→      const payload = (await response.json()) as { ok?: boolean; id?: string; error?: string };
   500→      if (!response.ok || !payload.ok) {
   501→        throw new Error(payload.error ?? 'Something went wrong.');
   502→      }
   503→      trackEvent('generate_lead', {
   504→        source,
   505→        service: serviceSlug,
   506→        estimate: estimate ? `${estimate.low}-${estimate.high}` : 'custom',
   507→        addons: pickedAddons.length,
   508→      });
   509→      setLeadId(payload.id ?? null);
   510→      setSubmitted(true);
   511→      toast({
   512→        title: 'Quote request received!',
   513→        description: 'Thanks — your fixed, itemized quote arrives within one business day.',
   514→      });
   515→    } catch (error) {
   516→      toast({
   517→        title: 'Could not send your request',
   518→        description: error instanceof Error ? error.message : 'Please try again or email us directly.',
   519→        variant: 'destructive',
   520→      });
   521→    } finally {
   522→      setSubmitting(false);
   523→    }
   524→  };
   525→
   526→  const resetToForm = () => {
   527→    setSubmitted(false);
   528→    setLeadId(null);
   529→    setErrors({});
   530→    setForm(EMPTY_FORM);
   531→  };
   532→
   533→  /* --------------------------- styles ----------------------------- */
   534→
   535→  const inputClass =
   536→    'h-11 rounded-xl border-gray-200 bg-white text-[#0a0a0a] placeholder:text-gray-400 focus-visible:border-emerald-500 focus-visible:ring-emerald-500';
   537→
   538→  /* -------------------------- sub-blocks -------------------------- */
   539→
   540→  const servicePicker = (
   541→    <ChoiceGroup step="Step 1" icon={Sparkles} title="What do you need?" hint="Pick a service — the quote builder adapts to it instantly.">
   542→      <div role="radiogroup" aria-label="Service needed" className="grid gap-2 sm:grid-cols-2">
   543→        {services.map((service) => (
   544→          <PillChoice
   545→            key={service.slug}
   546→            name={`service-${idPrefix}`}
   547→            value={service.slug}
   548→            idPrefix={`${idPrefix}-svc`}
   549→            selected={serviceSlug === service.slug}
   550→            onSelect={() => pickService(service.slug)}
   551→            label={service.shortName}
   552→            icon={<ServiceIconGlyph icon={service.icon} className="size-4" />}
   553→          />
   554→        ))}
   555→        <PillChoice
   556→          name={`service-${idPrefix}`}
   557→          value={NOT_SURE_SLUG}
   558→          idPrefix={`${idPrefix}-svc`}
