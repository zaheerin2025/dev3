   145→/** Quick slug → meta lookup. */
   146→export function getToolMeta(slug: string): ToolMeta | undefined {
   147→  return toolRegistry.find((tool) => tool.slug === slug);
   148→}
   149→
   150→/** Tools related to a slug: same category first, then neighbours. */
   151→export function getRelatedTools(slug: string, count = 4): ToolMeta[] {
   152→  const tool = getToolMeta(slug);
   153→  if (!tool) return [];
   154→  const sameCategory = toolRegistry.filter((x) => x.slug !== slug && x.category === tool.category);
   155→  const others = toolRegistry.filter((x) => x.slug !== slug && x.category !== tool.category);
   156→  return [...sameCategory, ...others].slice(0, count);
   157→}
