   880→interface MockupOpts {
   881→  frame: string; bgMode: 'transparent' | 'solid' | 'gradient'; bg1: string; bg2: string;
   882→  shadow: boolean; notch: boolean; scale: number;

   894→function drawMockup(canvas: HTMLCanvasElement, img: HTMLImageElement, o: MockupOpts) {
   895→  const s = o.scale;
   896→  const bodyH = 1400 * s;
   897→  const bezel = Math.max(12 * s, bodyH * 0.024);
   898→  const aspect = clamp((img.naturalWidth || 1) / (img.naturalHeight || 1), 0.45, 0.88);
   899→  const screenH = bodyH - bezel * 2;
   900→  const screenW = screenH * aspect;
   901→  const bodyW = screenW + bezel * 2;
   902→  const margin = (o.shadow ? bodyH * 0.075 : bodyH * 0.03);
   903→
   904→  canvas.width = Math.round(bodyW + margin * 2);
   905→  canvas.height = Math.round(bodyH + margin * 2);
   906→  const ctx = canvas.getContext('2d');
   907→  if (!ctx) return;
   908→
   909→  if (o.bgMode === 'solid') {
   910→    ctx.fillStyle = hexToRgb(o.bg1) ? o.bg1 : '#f4f1fa';
   911→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   912→  } else if (o.bgMode === 'gradient') {
   913→    const grad = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
   914→    grad.addColorStop(0, hexToRgb(o.bg1) ? o.bg1 : '#7c3aed');
   915→    grad.addColorStop(1, hexToRgb(o.bg2) ? o.bg2 : '#ec4899');
   916→    ctx.fillStyle = grad;
   917→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   918→  }
   919→
   920→  const bx = margin, by = margin;
   921→
   922→  if (o.shadow) {
   923→    ctx.save();
   924→    ctx.shadowColor = 'rgba(10, 10, 40, 0.45)';
   925→    ctx.shadowBlur = bodyH * 0.055;
   926→    ctx.shadowOffsetY = bodyH * 0.03;
   927→    ctx.fillStyle = o.frame;
   928→    roundRectPath(ctx, bx, by, bodyW, bodyH, bodyW * 0.16);
   929→    ctx.fill();
   930→    ctx.restore();
   931→  }
   932→
   933→  // body
   934→  ctx.fillStyle = o.frame;
   935→  roundRectPath(ctx, bx, by, bodyW, bodyH, bodyW * 0.16);
   936→  ctx.fill();
   937→  ctx.strokeStyle = 'rgba(255,255,255,0.14)';
   938→  ctx.lineWidth = 2 * s;
   939→  ctx.stroke();
   940→
   941→  // screen
   942→  const sx = bx + bezel, sy = by + bezel;
   943→  ctx.save();
   944→  roundRectPath(ctx, sx, sy, screenW, screenH, bodyW * 0.16 - bezel * 0.4);
   945→  ctx.clip();
   946→  ctx.fillStyle = '#0a0a0a';
   947→  ctx.fillRect(sx, sy, screenW, screenH);
   948→  drawImageCover(ctx, img, sx, sy, screenW, screenH);
   949→  ctx.restore();
   950→  ctx.strokeStyle = 'rgba(255,255,255,0.10)';
   951→  ctx.lineWidth = 1.5 * s;
   952→  roundRectPath(ctx, sx, sy, screenW, screenH, bodyW * 0.16 - bezel * 0.4);
   953→  ctx.stroke();
   954→
   955→  // notch
   956→  if (o.notch) {
   957→    ctx.fillStyle = darkened(o.frame, 0.35);
   958→    roundRectPath(ctx, bx + bodyW / 2 - screenW * 0.21, sy + bezel * 0.28, screenW * 0.42, bezel * 0.85, bezel * 0.42);
   959→    ctx.fill();
   960→  }
   961→
   962→  // side buttons
   963→  ctx.fillStyle = darkened(o.frame, 0.4);
   964→  roundRectPath(ctx, bx + bodyW, by + bodyH * 0.28, bezel * 0.38, bodyH * 0.10, bezel * 0.18);
   965→  ctx.fill(); // power
   966→  roundRectPath(ctx, bx - bezel * 0.38, by + bodyH * 0.22, bezel * 0.38, bodyH * 0.06, bezel * 0.18);
   967→  ctx.fill(); // vol up
   968→  roundRectPath(ctx, bx - bezel * 0.38, by + bodyH * 0.30, bezel * 0.38, bodyH * 0.06, bezel * 0.18);
   969→  ctx.fill(); // vol down
   970→}

  1089→  const maxY = Math.max(0, (dh - ch) / 2);

  1098→    ctx.strokeStyle = 'rgba(255, 255, 255, 0.75)';
  1099→    ctx.setLineDash([12, 10]);
  1100→    ctx.lineWidth = 3;
  1101→    ctx.beginPath(); ctx.moveTo(0, topH); ctx.lineTo(cw, topH); ctx.stroke();
  1102→    ctx.beginPath(); ctx.moveTo(0, ch - bottomH); ctx.lineTo(cw, ch - bottomH); ctx.stroke();
  1103→    ctx.setLineDash([]);
  1104→    ctx.fillStyle = 'rgba(255, 255, 255, 0.92)';
  1105→    ctx.font = `700 ${Math.round(cw * 0.032)}px ${CANVAS_FONT}`;
  1106→    ctx.textAlign = 'center';

  1108→    ctx.fillText('caption / engagement bar', cw / 2, ch - bottomH + cw * 0.06);
  1109→  }
