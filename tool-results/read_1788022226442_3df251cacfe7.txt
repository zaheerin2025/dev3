    64→    64→/** URL-safe slug: lowercase, alphanumerics and single dashes. */
    65→    65→function slugify(value: string): string {
    66→    66→  return value
    67→    67→    .toLowerCase()
    68→    68→    .replace(/[^a-z0-9]+/g, '-')
    69→    69→    .replace(/(^-|-$)+/g, '');
    70→    70→}

    72→    72→/**
    73→    73→ * Authorized fetch helper — the session rides in the HttpOnly cookie, so
    74→    74→ * same-origin requests attach it automatically. Throws on non-OK, parses
    75→    75→ * `{ ok, … }` JSON.
    76→    76→ */
    77→    77→async function adminFetch<T>(path: string, init?: RequestInit): Promise<T> {
    78→    78→  const response = await fetch(path, {
    79→    79→    ...init,
    80→    80→    headers: { 'Content-Type': 'application/json', ...(init?.headers ?? {}) },
    81→    81→  });
    82→    82→  const payload = (await response.json().catch(() => null)) as
    83→    83→    | ({ ok?: boolean; error?: string } & T)
    84→    84→    | null;
    85→    85→  if (!response.ok || !payload?.ok) {
    86→    86→    throw new Error(payload?.error ?? 'Request failed.');
    87→    87→  }
    88→    88→  return payload;
    89→    89→}

   114→   114→/* ─────────────────────────── Login gate ─────────────────────────── */
   115→   115→
   116→   116→function AdminLogin({ onAuthenticated }: { onAuthenticated: () => void }) {
   117→   117→  const { toast } = useToast();
   118→   118→  const [passcode, setPasscode] = React.useState('');
   119→   119→  const [loading, setLoading] = React.useState(false);
   120→   120→
   121→   121→  const submit = async (event: React.FormEvent) => {
   122→   122→    event.preventDefault();
   123→   123→    if (!passcode.trim() || loading) return;
   124→   124→    setLoading(true);
   125→   125→    try {
   126→   126→      const response = await fetch('/api/admin/auth', {
   127→   127→        method: 'POST',
   128→   128→        headers: { 'Content-Type': 'application/json' },
   129→   129→        body: JSON.stringify({ passcode: passcode.trim() }),
   130→   130→      });
   131→   131→      const payload = (await response.json().catch(() => null)) as {
   132→   132→        ok?: boolean;
   133→   133→        error?: string;
   134→   134→      } | null;
   135→   135→      if (!response.ok || !payload?.ok) {
   136→   136→        throw new Error(payload?.error ?? 'Invalid passcode.');
   137→   137→      }
   138→   138→      onAuthenticated();
   139→   139→    } catch (error) {
   140→   140→      toast({
   141→   141→        title: 'Sign in failed',
   142→   142→        description: error instanceof Error ? error.message : 'Please try again.',
   143→   143→        variant: 'destructive',
   144→   144→      });
   145→   145→    } finally {
   146→   146→      setLoading(false);
   147→   147→    }
   148→   148→  };
   149→   149→
   150→   150→  return (
   151→   151→    <div className="flex min-h-[60vh] items-center justify-center py-16">
   152→   152→      <div className="card-surface w-full max-w-sm rounded-3xl p-8 shadow-[0_24px_48px_-24px_rgb(4_16_11/0.25)]">
   153→   153→        <div className="flex flex-col items-center gap-1.5 text-center">
   154→   154→          <img src="/logo.svg" alt="Developers3 logo" className="h-10 w-10" />
   155→   155→          <h1 className="mt-2 text-xl font-bold tracking-tight">Admin panel</h1>
   156→   156→          <p className="text-sm text-muted-foreground">
   157→   157→            Sign in to manage posts, pricing, and site content.
   158→   158→          </p>
   159→   159→        </div>
   160→   160→        <form onSubmit={submit} className="mt-6 flex flex-col gap-4">
   161→   161→          <div className="flex flex-col gap-2">
   162→   162→            <Label htmlFor="admin-passcode">Passcode</Label>
   163→   163→            <Input
   164→   164→              id="admin-passcode"
   165→   165→              type="password"
   166→   166→              value={passcode}
   167→   167→              onChange={(event) => setPasscode(event.target.value)}
   168→   168→              placeholder="Enter the admin passcode"
   169→   169→              autoComplete="current-password"
   170→   170→              autoFocus
   171→   171→              required
   172→   172→            />
   173→   173→          </div>
   174→   174→          <Button type="submit" className="h-11 w-full" disabled={loading}>
   175→   175→            {loading ? (
   176→   176→              <>
   177→   177→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   178→   178→                Signing in…
   179→   179→              </>
   180→   180→            ) : (
   181→   181→              'Sign in'
   182→   182→            )}
   183→   183→          </Button>
   184→   184→        </form>
   185→   185→      </div>
   186→   186→    </div>
   187→   187→  );
   188→   188→}
