    58→    58→/** URL-safe slug: lowercase, alphanumerics and single dashes. */
    59→    59→function slugify(value: string): string {
    60→    60→  return value
    61→    61→    .toLowerCase()
    62→    62→    .replace(/[^a-z0-9]+/g, '-')
    63→    63→    .replace(/(^-|-$)+/g, '');
    64→    64→}

    66→    66→/** Authorized fetch helper — throws on non-OK, parses `{ ok, … }` JSON. */
    67→    67→async function adminFetch<T>(path: string, token: string, init?: RequestInit): Promise<T> {
    68→    68→  const response = await fetch(path, {
    69→    69→    ...init,
    70→    70→    headers: { 'Content-Type': 'application/json', 'x-admin-token': token },
    71→    71→  });
    72→    72→  const payload = (await response.json().catch(() => null)) as
    73→    73→    | ({ ok?: boolean; error?: string } & T)
    74→    74→    | null;
    75→    75→  if (!response.ok || !payload?.ok) {
    76→    76→    throw new Error(payload?.error ?? 'Request failed.');
    77→    77→  }
    78→    78→  return payload;
    79→    79→}

   105→   105→
   106→   106→function AdminLogin({ onAuthenticated }: { onAuthenticated: (token: string) => void }) {
   107→   107→  const { toast } = useToast();
   108→   108→  const [passcode, setPasscode] = React.useState('');
   109→   109→  const [loading, setLoading] = React.useState(false);
   110→   110→
   111→   111→  const submit = async (event: React.FormEvent) => {
   112→   112→    event.preventDefault();
   113→   113→    if (!passcode.trim() || loading) return;
   114→   114→    setLoading(true);
   115→   115→    try {
   116→   116→      const response = await fetch('/api/admin/auth', {
   117→   117→        method: 'POST',
   118→   118→        headers: { 'Content-Type': 'application/json' },
   119→   119→        body: JSON.stringify({ passcode: passcode.trim() }),
   120→   120→      });
   121→   121→      const payload = (await response.json().catch(() => null)) as {
   122→   122→        ok?: boolean;
   123→   123→        token?: string;
   124→   124→        error?: string;
   125→   125→      } | null;
   126→   126→      if (!response.ok || !payload?.ok || !payload.token) {
   127→   127→        throw new Error(payload?.error ?? 'Invalid passcode.');
   128→   128→      }
   129→   129→      localStorage.setItem(TOKEN_KEY, payload.token);
   130→   130→      onAuthenticated(payload.token);
   131→   131→    } catch (error) {
   132→   132→      toast({
   133→   133→        title: 'Sign in failed',
   134→   134→        description: error instanceof Error ? error.message : 'Please try again.',
   135→   135→        variant: 'destructive',
   136→   136→      });
   137→   137→    } finally {
   138→   138→      setLoading(false);
   139→   139→    }
   140→   140→  };
   141→   141→
   142→   142→  return (
   143→   143→    <div className="flex min-h-[60vh] items-center justify-center py-16">
   144→   144→      <div className="card-surface w-full max-w-sm rounded-3xl p-8 shadow-[0_24px_48px_-24px_rgb(4_16_11/0.25)]">
   145→   145→        <div className="flex flex-col items-center gap-1.5 text-center">
   146→   146→          <img src="/logo.svg" alt="Developers3 logo" className="h-10 w-10" />
   147→   147→          <h1 className="mt-2 text-xl font-bold tracking-tight">Admin panel</h1>
   148→   148→          <p className="text-sm text-muted-foreground">
   149→   149→            Sign in to manage posts, pricing, and site content.
   150→   150→          </p>
   151→   151→        </div>
   152→   152→        <form onSubmit={submit} className="mt-6 flex flex-col gap-4">
   153→   153→          <div className="flex flex-col gap-2">
   154→   154→            <Label htmlFor="admin-passcode">Passcode</Label>
   155→   155→            <Input
   156→   156→              id="admin-passcode"
   157→   157→              type="password"
   158→   158→              value={passcode}
   159→   159→              onChange={(event) => setPasscode(event.target.value)}
   160→   160→              placeholder="Enter the admin passcode"
   161→   161→              autoComplete="current-password"
   162→   162→              autoFocus
   163→   163→              required
   164→   164→            />
   165→   165→          </div>
   166→   166→          <Button type="submit" className="h-11 w-full" disabled={loading}>
   167→   167→            {loading ? (
   168→   168→              <>
   169→   169→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   170→   170→                Signing in…
   171→   171→              </>
   172→   172→            ) : (
   173→   173→              'Sign in'
   174→   174→            )}
   175→   175→          </Button>
   176→   176→        </form>
   177→   177→      </div>
   178→   178→    </div>
   179→   179→  );
   180→   180→}
