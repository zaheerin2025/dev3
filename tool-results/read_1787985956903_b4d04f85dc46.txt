    90→function Field({
    91→  id,
    92→  label,
    93→  required,
    94→  optional,
    95→  error,
    96→  className,
    97→  children,
    98→}: {
    99→  id: string;
   100→  label: string;
   101→  required?: boolean;
   102→  optional?: boolean;
   103→  error?: string;
   104→  className?: string;
   105→  children: React.ReactNode;
   106→}) {
   107→  return (
   108→    <div className={cn('flex flex-col gap-1.5', className)}>
   109→      <Label htmlFor={id} className="text-sm font-bold text-[#0a0a0a]">
   110→        {label}
   111→        {required ? (
   112→          <span className="ml-0.5 text-purple-600" aria-hidden="true">
   113→            *
   114→          </span>
   115→        ) : null}
   116→        {optional ? <span className="ml-1 text-xs font-normal text-muted-foreground">(optional)</span> : null}
   117→      </Label>
   118→      {children}
   119→      {error ? (
   120→        <p id={`${id}-error`} className="mt-1 text-xs font-medium text-destructive">
   121→          {error}
   122→        </p>
   123→      ) : null}
   124→    </div>
   125→  );
   126→}

   197→/**
   198→ * Project inquiry form with honeypot spam protection.
   199→ * Posts to /api/leads (stored in DB) and fires a GA4 generate_lead event.
   200→ */
   201→export function LeadForm({
   202→  source = 'contact',
   203→  defaultService,
   204→  dark,
   205→  compact,
   206→  bare,
   207→  className,
   208→}: LeadFormProps) {
   209→  const { toast } = useToast();
   210→  const [form, setForm] = React.useState<FormState>(() => ({
   211→    ...EMPTY_FORM,
   212→    service: defaultService ?? '',
   213→  }));
   214→  const [honeypot, setHoneypot] = React.useState('');
   215→  const [errors, setErrors] = React.useState<Errors>({});
   216→  const [submitting, setSubmitting] = React.useState(false);
   217→  const [submitted, setSubmitted] = React.useState(false);
   218→  const [leadId, setLeadId] = React.useState<string | null>(null);
   219→
   220→  /** Map form fields to their validation error keys. */
   221→  const ERROR_KEY_FOR: Partial<Record<keyof FormState, keyof Errors>> = {
   222→    firstName: 'name',
   223→    email: 'email',
   224→    service: 'service',
   225→    budget: 'budget',
   226→    message: 'message',
   227→  };
   228→
   229→  const set = (key: keyof FormState) => (value: string) => {
   230→    setForm((prev) => ({ ...prev, [key]: value }));
   231→    const errorKey = ERROR_KEY_FOR[key];
   232→    if (!errorKey) return;
   233→    setErrors((prev) => (prev[errorKey] ? { ...prev, [errorKey]: undefined } : prev));
   234→  };
   235→
   236→  const validate = (): boolean => {
   237→    const next: Errors = {};
   238→    if (form.firstName.trim().length < 2) next.name = 'Please enter your first name.';
   239→    if (!EMAIL_REGEX.test(form.email.trim())) next.email = 'Please enter a valid email address.';
   240→    if (!form.service) next.service = 'Please select a service.';
   241→    if (!form.budget) next.budget = 'Please select a budget range.';
   242→    if (form.message.trim().length < MESSAGE_MIN) {
   243→      next.message = `Please add a little more detail (at least ${MESSAGE_MIN} characters).`;
   244→    }
   245→    setErrors(next);
   246→    return Object.keys(next).length === 0;
   247→  };
   248→
   249→  const handleSubmit = async (event: React.FormEvent) => {
   250→    event.preventDefault();
   251→    if (!validate()) return;
   252→    setSubmitting(true);
   253→    try {
   254→      const fullName = [form.firstName.trim(), form.lastName.trim()].filter(Boolean).join(' ');
   255→      // Company is folded into the message so nothing the client shares gets lost.
   256→      const message = form.company.trim()
   257→        ? `Company: ${form.company.trim()}\n\n${form.message}`
   258→        : form.message;
   259→
   260→      const response = await fetch('/api/leads', {
   261→        method: 'POST',
   262→        headers: { 'Content-Type': 'application/json' },
   263→        body: JSON.stringify({
   264→          name: fullName,
   265→          email: form.email,
   266→          phone: form.phone,
   267→          service: form.service,
   268→          budget: form.budget,
   269→          timeline: form.timeline,
   270→          message,
   271→          source,
   272→          website: honeypot,
   273→        }),
   274→      });
   275→      const payload = (await response.json()) as { ok?: boolean; id?: string; error?: string };
   276→      if (!response.ok || !payload.ok) {
   277→        throw new Error(payload.error ?? 'Something went wrong.');
   278→      }
   279→      trackEvent('generate_lead', {
   280→        source,
   281→        service: form.service || 'unspecified',
   282→        budget: form.budget || 'unspecified',
   283→        timeline: form.timeline || 'unspecified',
   284→      });
   285→      setLeadId(payload.id ?? null);
   286→      setSubmitted(true);
   287→      setForm({ ...EMPTY_FORM, service: defaultService ?? '' });
   288→      toast({
   289→        title: 'Request received!',
   290→        description: 'Thanks — we will send your fixed quote within one business day.',
   291→      });
   292→    } catch (error) {
   293→      toast({
   294→        title: 'Could not send your request',
   295→        description: error instanceof Error ? error.message : 'Please try again or email us directly.',
   296→        variant: 'destructive',
   297→      });
   298→    } finally {
   299→      setSubmitting(false);
   300→    }
   301→  };
   302→
   303→  const resetToForm = () => {
   304→    setSubmitted(false);
   305→    setLeadId(null);
   306→    setErrors({});
   307→  };
   308→
   309→  /* ---------------------------- styles ---------------------------- */
   310→
   311→  // The split layout pairs a dark rail with a white form (pricing / service
   312→  // pages, full content width). `bare` is the clear single-column form —
   313→  // the parent (contact page gradient-border card) supplies the surface.
   314→  const wrapperClass = bare
   315→    ? 'rounded-[24px]'
   316→    : cn('card-soft grid overflow-hidden rounded-[24px] lg:grid-cols-[0.85fr_1.15fr]');
   317→  const inputClass = dark
   318→    ? 'h-11 rounded-xl border-white/15 bg-white/5 text-white placeholder:text-white/40 focus-visible:border-purple-400 focus-visible:ring-purple-400'
   319→    : 'h-11 rounded-xl border-gray-200 bg-white text-[#0a0a0a] placeholder:text-gray-400 focus-visible:border-purple-500 focus-visible:ring-purple-500';
   320→  const triggerClass = cn(inputClass, 'w-full data-[size=default]:h-11');
   321→
   322→  /* ------------------------- form fields -------------------------- */
   323→
   324→  const timelineField = (
   325→    <Field id={`lead-timeline-${source}`} label="Timeline" optional>
   326→      <Select value={form.timeline} onValueChange={set('timeline')}>
   327→        <SelectTrigger
   328→          id={`lead-timeline-${source}`}
   329→          className={triggerClass}
   330→          aria-label="Project timeline"
   331→        >
   332→          <SelectValue placeholder="When do you want to start?" />
   333→        </SelectTrigger>
   334→        <SelectContent>
   335→          {TIMELINE_OPTIONS.map((option) => (
   336→            <SelectItem key={option} value={option}>
   337→              {option}
   338→            </SelectItem>
   339→          ))}
   340→        </SelectContent>
   341→      </Select>
   342→    </Field>
   343→  );
   344→
   345→  const companyField = (
   346→    <Field id={`lead-company-${source}`} label="Company" optional>
   347→      <Input
   348→        id={`lead-company-${source}`}
   349→        value={form.company}
   350→        onChange={(e) => set('company')(e.target.value)}
   351→        placeholder="Company or brand name"
   352→        autoComplete="organization"
   353→        className={inputClass}
   354→      />
   355→    </Field>
   356→  );
   357→
   358→  const nameFields = (
   359→    <>
   360→      <Field
   361→        id={`lead-first-${source}`}
   362→        label="First name"
   363→        required
   364→        error={errors.name}
   365→      >
   366→        <Input
   367→          id={`lead-first-${source}`}
   368→          value={form.firstName}
   369→          onChange={(e) => set('firstName')(e.target.value)}
   370→          placeholder="Jane"
   371→          autoComplete="given-name"
   372→          required
   373→          aria-invalid={Boolean(errors.name)}
   374→          aria-describedby={errors.name ? `lead-first-${source}-error` : undefined}
   375→          className={inputClass}
   376→        />
   377→      </Field>
   378→      <Field id={`lead-last-${source}`} label="Last name" optional>
   379→        <Input
   380→          id={`lead-last-${source}`}
   381→          value={form.lastName}
   382→          onChange={(e) => set('lastName')(e.target.value)}
   383→          placeholder="Smith"
   384→          autoComplete="family-name"
   385→          className={inputClass}
   386→        />
   387→      </Field>
   388→    </>
   389→  );
   390→
   391→  const contactFields = (
   392→    <>
   393→      <Field id={`lead-email-${source}`} label="Email" required error={errors.email}>
   394→        <Input
   395→          id={`lead-email-${source}`}
   396→          type="email"
   397→          value={form.email}
   398→          onChange={(e) => set('email')(e.target.value)}
   399→          placeholder="jane@company.com"
   400→          autoComplete="email"
   401→          required
   402→          aria-invalid={Boolean(errors.email)}
   403→          aria-describedby={errors.email ? `lead-email-${source}-error` : undefined}
   404→          className={inputClass}
   405→        />
   406→      </Field>
   407→      <Field id={`lead-phone-${source}`} label="Phone" optional>
   408→        <Input
   409→          id={`lead-phone-${source}`}
   410→          type="tel"
   411→          value={form.phone}
   412→          onChange={(e) => set('phone')(e.target.value)}
   413→          placeholder="+1 (555) 000-0000"
   414→          autoComplete="tel"
   415→          className={inputClass}
   416→        />
   417→      </Field>
   418→    </>
   419→  );
   420→
   421→  const pickFields = (
   422→    <>
   423→      <Field id={`lead-service-${source}`} label="Service" required error={errors.service}>
   424→        <Select value={form.service} onValueChange={set('service')}>
   425→          <SelectTrigger
   426→            id={`lead-service-${source}`}
   427→            className={triggerClass}
   428→            aria-label="Service needed"
   429→            aria-invalid={Boolean(errors.service)}
   430→            aria-describedby={errors.service ? `lead-service-${source}-error` : undefined}
   431→          >
   432→            <SelectValue placeholder="What do you need?" />
   433→          </SelectTrigger>
   434→          <SelectContent>
   435→            {services.map((service) => (
   436→              <SelectItem key={service.slug} value={service.slug}>
   437→                {service.name}
   438→              </SelectItem>
   439→            ))}
   440→            <SelectItem value="not-sure">Not sure yet / something else</SelectItem>
   441→          </SelectContent>
   442→        </Select>
   443→      </Field>
   444→      <Field id={`lead-budget-${source}`} label="Budget" required error={errors.budget}>
   445→        <Select value={form.budget} onValueChange={set('budget')}>
   446→          <SelectTrigger
   447→            id={`lead-budget-${source}`}
   448→            className={triggerClass}
   449→            aria-label="Budget range"
   450→            aria-invalid={Boolean(errors.budget)}
   451→            aria-describedby={errors.budget ? `lead-budget-${source}-error` : undefined}
   452→          >
   453→            <SelectValue placeholder="Select a range" />
   454→          </SelectTrigger>
   455→          <SelectContent>
   456→            {BUDGET_RANGES.map((range) => (
   457→              <SelectItem key={range} value={range}>
   458→                {range}
   459→              </SelectItem>
   460→            ))}
   461→          </SelectContent>
   462→        </Select>
   463→      </Field>
   464→    </>
   465→  );
   466→
   467→  const messageField = (
   468→    <Field
   469→      id={`lead-message-${source}`}
   470→      label="Project details"
   471→      required
   472→      error={errors.message}
   473→      className={compact ? undefined : 'sm:col-span-2'}
   474→    >
   475→      <Textarea
   476→        id={`lead-message-${source}`}
   477→        value={form.message}
   478→        onChange={(e) => set('message')(e.target.value)}
   479→        placeholder="Tell us about your project, goals, and timeline…"
   480→        rows={compact ? 4 : 5}
   481→        maxLength={MESSAGE_MAX}
   482→        required
   483→        aria-invalid={Boolean(errors.message)}
   484→        aria-describedby={
   485→          errors.message ? `lead-message-${source}-error` : `lead-message-${source}-hint`
   486→        }
   487→        className={cn(
   488→          inputClass,
   489→          'h-auto min-h-[120px] py-3 focus-visible:border-purple-500 focus-visible:ring-purple-500'
   490→        )}
   491→      />
   492→      <div
   493→        id={`lead-message-${source}-hint`}
   494→        className="flex items-center justify-between text-[11px] text-muted-foreground"
   495→      >
   496→        <span>Min. {MESSAGE_MIN} characters — goals, scope, links all help.</span>
   497→        <span
   498→          className={cn('tabular-nums', form.message.length >= MESSAGE_MAX - 20 && 'text-destructive')}
   499→          aria-live="polite"
   500→        >
   501→          {form.message.length}/{MESSAGE_MAX}
   502→        </span>
   503→      </div>
   504→    </Field>
   505→  );
   506→
   507→  const honeypotField = (
   508→    <div className="absolute -left-[9999px] top-auto h-px w-px overflow-hidden" aria-hidden="true">
   509→      <label>
   510→        Company website
   511→        <input
   512→          type="text"
   513→          name="website"
   514→          tabIndex={-1}
   515→          autoComplete="off"
   516→          value={honeypot}
   517→          onChange={(e) => setHoneypot(e.target.value)}
   518→        />
   519→      </label>
   520→    </div>
   521→  );
   522→
   523→  const consentCopy = (
   524→    <p
   525→      className={cn(
   526→        'text-xs leading-relaxed',
   527→        dark ? 'text-slate-400' : 'text-muted-foreground'
   528→      )}
   529→    >
   530→      By submitting, you agree to our{' '}
   531→      <Link
   532→        href="/privacy-policy"
   533→        className="font-medium text-purple-700 underline decoration-purple-300 underline-offset-2 hover:text-purple-800"
   534→      >
   535→        Privacy Policy
   536→      </Link>
   537→      . We reply within one business day and never share your details.
   538→    </p>
   539→  );
   540→
   541→  const submitButton = (
   542→    <button
   543→      type="submit"
   544→      disabled={submitting}
   545→      className="btn-primary-pill w-full disabled:pointer-events-none disabled:opacity-60"
   546→    >
   547→      {submitting ? (
   548→        <>
   549→          <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   550→          Sending…
   551→        </>
   552→      ) : (
   553→        <>
   554→          <Send className="h-4 w-4" aria-hidden="true" />
   555→          Send message
   556→        </>
   557→      )}
   558→    </button>
   559→  );
   560→
   561→  /* -------------------------- success ----------------------------- */
   562→
   563→  if (submitted) {
   564→    if (compact) {
   565→      return (
   566→        <div
   567→          className={cn('card-soft overflow-hidden rounded-[24px]', className)}
   568→          data-slot="lead-form-success"
   569→        >
   570→          <SuccessPanel leadId={leadId} dark={dark} onReset={resetToForm} />
   571→        </div>
   572→      );
   573→    }
   574→    if (bare) {
   575→      return (
   576→        <div
   577→          className={cn('relative rounded-[24px] bg-white p-6 sm:p-8 lg:p-10', className)}
   578→          data-slot="lead-form-success"
   579→        >
   580→          <SuccessPanel leadId={leadId} dark={dark} onReset={resetToForm} />
   581→        </div>
   582→      );
   583→    }
   584→    return (
   585→      <div
   586→        className={cn(wrapperClass, className)}
   587→        data-slot="lead-form-success"
   588→      >
   589→        <LeadRail />
   590→        <div className="relative bg-white p-6 sm:p-8 lg:p-10">
   591→          <SuccessPanel leadId={leadId} dark={dark} onReset={resetToForm} />
   592→        </div>
   593→      </div>
   594→    );
   595→  }
   596→
   597→  /* ------------------------------ bare ----------------------------- */
   598→  /*
   599→   * `bare` — the CLEAR single-column form used inside the contact page's
   600→   * gradient-border card. No dark rail, no nested splits: one white,
   601→   * airy, readable form with its own heading. Parent supplies the card.
   602→   */
   603→  if (bare) {
   604→    return (
   605→      <div className={cn('relative bg-white p-6 sm:p-8 lg:p-10', className)} data-slot="lead-form">
   606→        {submitted ? (
   607→          <SuccessPanel leadId={leadId} dark={dark} onReset={resetToForm} />
