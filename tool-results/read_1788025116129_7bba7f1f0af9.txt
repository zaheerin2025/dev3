    85→function Field({
    86→  id,
    87→  label,
    88→  required,
    89→  optional,
    90→  error,
    91→  className,
    92→  children,
    93→}: {
    94→  id: string;
    95→  label: string;
    96→  required?: boolean;
    97→  optional?: boolean;
    98→  error?: string;
    99→  className?: string;
   100→  children: React.ReactNode;
   101→}) {
   102→  return (
   103→    <div className={cn('flex flex-col gap-1.5', className)}>
   104→      <Label htmlFor={id} className="text-sm font-bold text-[#0a0a0a]">
   105→        {label}
   106→        {required ? (
   107→          <span className="ml-0.5 text-purple-600" aria-hidden="true">
   108→            *
   109→          </span>
   110→        ) : null}
   111→        {optional ? <span className="ml-1 text-xs font-normal text-muted-foreground">(optional)</span> : null}
   112→      </Label>
   113→      {children}
   114→      {error ? (
   115→        <p id={`${id}-error`} className="mt-1 text-xs font-medium text-destructive">
   116→          {error}
   117→        </p>
   118→      ) : null}
   119→    </div>
   120→  );
   121→}

   194→/**
   195→ * Project inquiry form with honeypot spam protection.
   196→ * Posts to /api/leads (stored in DB) and fires a GA4 generate_lead event.
   197→ */
   198→export function LeadForm({
   199→  source = 'contact',
   200→  defaultService,
   201→  dark,
   202→  compact,
   203→  bare,
   204→  className,
   205→}: LeadFormProps) {
   206→  const { toast } = useToast();
   207→  const [form, setForm] = React.useState<FormState>(() => ({
   208→    ...EMPTY_FORM,
   209→    service: defaultService ?? '',
   210→  }));
   211→  const [honeypot, setHoneypot] = React.useState('');
   212→  const [errors, setErrors] = React.useState<Errors>({});
   213→  const [submitting, setSubmitting] = React.useState(false);
   214→  const [submitted, setSubmitted] = React.useState(false);
   215→  const [leadId, setLeadId] = React.useState<string | null>(null);
   216→
   217→  /** Map form fields to their validation error keys. */
   218→  const ERROR_KEY_FOR: Partial<Record<keyof FormState, keyof Errors>> = {
   219→    firstName: 'name',
   220→    email: 'email',
   221→    service: 'service',
   222→    budget: 'budget',
   223→    message: 'message',
   224→  };
   225→
   226→  const set = (key: keyof FormState) => (value: string) => {
   227→    setForm((prev) => ({ ...prev, [key]: value }));
   228→    const errorKey = ERROR_KEY_FOR[key];
   229→    if (!errorKey) return;
   230→    setErrors((prev) => (prev[errorKey] ? { ...prev, [errorKey]: undefined } : prev));
   231→  };
   232→
   233→  const validate = (): boolean => {
   234→    const next: Errors = {};
   235→    if (form.firstName.trim().length < 2) next.name = 'Please enter your first name.';
   236→    if (!EMAIL_REGEX.test(form.email.trim())) next.email = 'Please enter a valid email address.';
   237→    if (!form.service) next.service = 'Please select a service.';
   238→    if (!form.budget) next.budget = 'Please select a budget range.';
   239→    if (form.message.trim().length < MESSAGE_MIN) {
   240→      next.message = `Please add a little more detail (at least ${MESSAGE_MIN} characters).`;
   241→    }
   242→    setErrors(next);
   243→    return Object.keys(next).length === 0;
   244→  };
   245→
   246→  const handleSubmit = async (event: React.FormEvent) => {
   247→    event.preventDefault();
   248→    if (!validate()) return;
   249→    setSubmitting(true);
   250→    try {
   251→      const fullName = [form.firstName.trim(), form.lastName.trim()].filter(Boolean).join(' ');
   252→      // Company is folded into the message so nothing the client shares gets lost.
   253→      const message = form.company.trim()
   254→        ? `Company: ${form.company.trim()}\n\n${form.message}`
   255→        : form.message;
   256→
   257→      const response = await fetch('/api/leads', {
   258→        method: 'POST',
   259→        headers: { 'Content-Type': 'application/json' },
   260→        body: JSON.stringify({
   261→          name: fullName,
   262→          email: form.email,
   263→          phone: form.phone,
   264→          service: form.service,
   265→          budget: form.budget,
   266→          timeline: form.timeline,
   267→          message,
   268→          source,
   269→          website: honeypot,
   270→        }),
   271→      });
   272→      const payload = (await response.json()) as { ok?: boolean; id?: string; error?: string };
   273→      if (!response.ok || !payload.ok) {
   274→        throw new Error(payload.error ?? 'Something went wrong.');
   275→      }
   276→      trackEvent('generate_lead', {
   277→        source,
   278→        service: form.service || 'unspecified',
   279→        budget: form.budget || 'unspecified',
   280→        timeline: form.timeline || 'unspecified',
   281→      });
   282→      setLeadId(payload.id ?? null);
   283→      setSubmitted(true);
   284→      setForm({ ...EMPTY_FORM, service: defaultService ?? '' });
   285→      toast({
   286→        title: 'Request received!',
   287→        description: 'Thanks — we will send your fixed quote within one business day.',
   288→      });
   289→    } catch (error) {
   290→      toast({
   291→        title: 'Could not send your request',
   292→        description: error instanceof Error ? error.message : 'Please try again or email us directly.',
   293→        variant: 'destructive',
   294→      });
   295→    } finally {
   296→      setSubmitting(false);
   297→    }
   298→  };
   299→
   300→  const resetToForm = () => {
   301→    setSubmitted(false);
   302→    setLeadId(null);
   303→    setErrors({});
   304→  };
   305→
   306→  /* ---------------------------- styles ---------------------------- */
   307→
   308→  // The split layout pairs a dark rail with a white form (pricing / service
   309→  // pages, full content width). `bare` is the clear single-column form —
   310→  // the parent (contact page gradient-border card) supplies the surface.
   311→  const wrapperClass = bare
   312→    ? 'rounded-[24px]'
   313→    : cn('card-soft grid overflow-hidden rounded-[24px] lg:grid-cols-[0.85fr_1.15fr]');
   314→  const inputClass = dark
   315→    ? 'h-11 rounded-xl border-white/15 bg-white/5 text-white placeholder:text-white/40 focus-visible:border-purple-400 focus-visible:ring-purple-400'
   316→    : 'h-11 rounded-xl border-gray-200 bg-white text-[#0a0a0a] placeholder:text-gray-400 focus-visible:border-purple-500 focus-visible:ring-purple-500';
   317→  const triggerClass = cn(inputClass, 'w-full data-[size=default]:h-11');
   318→
   319→  /* ------------------------- form fields -------------------------- */
   320→
   321→  const timelineField = (
   322→    <Field id={`lead-timeline-${source}`} label="Timeline" optional>
   323→      <Select value={form.timeline} onValueChange={set('timeline')}>
   324→        <SelectTrigger
   325→          id={`lead-timeline-${source}`}
   326→          className={triggerClass}
   327→          aria-label="Project timeline"
   328→        >
   329→          <SelectValue placeholder="When do you want to start?" />
   330→        </SelectTrigger>
   331→        <SelectContent>
   332→          {TIMELINE_OPTIONS.map((option) => (
   333→            <SelectItem key={option} value={option}>
   334→              {option}
   335→            </SelectItem>
   336→          ))}
   337→        </SelectContent>
   338→      </Select>
   339→    </Field>
   340→  );
   341→
   342→  const companyField = (
   343→    <Field id={`lead-company-${source}`} label="Company" optional>
   344→      <Input
   345→        id={`lead-company-${source}`}
   346→        value={form.company}
   347→        onChange={(e) => set('company')(e.target.value)}
   348→        placeholder="Company or brand name"
   349→        autoComplete="organization"
   350→        className={inputClass}
   351→      />
   352→    </Field>
   353→  );
   354→
   355→  const nameFields = (
   356→    <>
   357→      <Field
   358→        id={`lead-first-${source}`}
   359→        label="First name"
   360→        required
   361→        error={errors.name}
   362→      >
   363→        <Input
   364→          id={`lead-first-${source}`}
   365→          value={form.firstName}
   366→          onChange={(e) => set('firstName')(e.target.value)}
   367→          placeholder="Jane"
   368→          autoComplete="given-name"
   369→          required
   370→          aria-invalid={Boolean(errors.name)}
   371→          aria-describedby={errors.name ? `lead-first-${source}-error` : undefined}
   372→          className={inputClass}
   373→        />
   374→      </Field>
   375→      <Field id={`lead-last-${source}`} label="Last name" optional>
   376→        <Input
   377→          id={`lead-last-${source}`}
   378→          value={form.lastName}
   379→          onChange={(e) => set('lastName')(e.target.value)}
   380→          placeholder="Smith"
   381→          autoComplete="family-name"
   382→          className={inputClass}
   383→        />
   384→      </Field>
   385→    </>
   386→  );
   387→
   388→  const contactFields = (
   389→    <>
   390→      <Field id={`lead-email-${source}`} label="Email" required error={errors.email}>
   391→        <Input
   392→          id={`lead-email-${source}`}
   393→          type="email"
   394→          value={form.email}
   395→          onChange={(e) => set('email')(e.target.value)}
   396→          placeholder="jane@company.com"
   397→          autoComplete="email"
   398→          required
   399→          aria-invalid={Boolean(errors.email)}
   400→          aria-describedby={errors.email ? `lead-email-${source}-error` : undefined}
   401→          className={inputClass}
   402→        />
   403→      </Field>
   404→      <Field id={`lead-phone-${source}`} label="Phone" optional>
   405→        <Input
   406→          id={`lead-phone-${source}`}
   407→          type="tel"
   408→          value={form.phone}
   409→          onChange={(e) => set('phone')(e.target.value)}
   410→          placeholder="+1 (555) 000-0000"
   411→          autoComplete="tel"
   412→          className={inputClass}
   413→        />
   414→      </Field>
   415→    </>
   416→  );
   417→
   418→  const pickFields = (
   419→    <>
   420→      <Field id={`lead-service-${source}`} label="Service" required error={errors.service}>
   421→        <Select value={form.service} onValueChange={set('service')}>
   422→          <SelectTrigger
   423→            id={`lead-service-${source}`}
   424→            className={triggerClass}
   425→            aria-label="Service needed"
   426→            aria-invalid={Boolean(errors.service)}
   427→            aria-describedby={errors.service ? `lead-service-${source}-error` : undefined}
   428→          >
   429→            <SelectValue placeholder="What do you need?" />
   430→          </SelectTrigger>
   431→          <SelectContent>
   432→            {services.map((service) => (
   433→              <SelectItem key={service.slug} value={service.slug}>
   434→                {service.name}
   435→              </SelectItem>
   436→            ))}
   437→            <SelectItem value="not-sure">Not sure yet / something else</SelectItem>
   438→          </SelectContent>
   439→        </Select>
   440→      </Field>
   441→      <Field id={`lead-budget-${source}`} label="Budget" required error={errors.budget}>
   442→        <Select value={form.budget} onValueChange={set('budget')}>
   443→          <SelectTrigger
   444→            id={`lead-budget-${source}`}
   445→            className={triggerClass}
   446→            aria-label="Budget range"
   447→            aria-invalid={Boolean(errors.budget)}
   448→            aria-describedby={errors.budget ? `lead-budget-${source}-error` : undefined}
   449→          >
   450→            <SelectValue placeholder="Select a range" />
   451→          </SelectTrigger>
   452→          <SelectContent>
   453→            {BUDGET_RANGES.map((range) => (
   454→              <SelectItem key={range} value={range}>
   455→                {range}
   456→              </SelectItem>
   457→            ))}
   458→          </SelectContent>
   459→        </Select>
   460→      </Field>
   461→    </>
   462→  );
   463→
   464→  const messageField = (
   465→    <Field
   466→      id={`lead-message-${source}`}
   467→      label="Project details"
   468→      required
   469→      error={errors.message}
   470→      className={compact ? undefined : 'sm:col-span-2'}
   471→    >
   472→      <Textarea
   473→        id={`lead-message-${source}`}
   474→        value={form.message}
   475→        onChange={(e) => set('message')(e.target.value)}
   476→        placeholder="Tell us about your project, goals, and timeline…"
   477→        rows={compact ? 4 : 5}
   478→        maxLength={MESSAGE_MAX}
   479→        required
   480→        aria-invalid={Boolean(errors.message)}
   481→        aria-describedby={
   482→          errors.message ? `lead-message-${source}-error` : `lead-message-${source}-hint`
   483→        }
   484→        className={cn(
   485→          inputClass,
   486→          'h-auto min-h-[120px] py-3 focus-visible:border-purple-500 focus-visible:ring-purple-500'
   487→        )}
   488→      />
   489→      <div
   490→        id={`lead-message-${source}-hint`}
   491→        className="flex items-center justify-between text-[11px] text-muted-foreground"
   492→      >
   493→        <span>Min. {MESSAGE_MIN} characters — goals, scope, links all help.</span>
   494→        <span
   495→          className={cn('tabular-nums', form.message.length >= MESSAGE_MAX - 20 && 'text-destructive')}
   496→          aria-live="polite"
   497→        >
   498→          {form.message.length}/{MESSAGE_MAX}
   499→        </span>
   500→      </div>
   501→    </Field>
   502→  );
   503→
   504→  const honeypotField = (
   505→    <div className="absolute -left-[9999px] top-auto h-px w-px overflow-hidden" aria-hidden="true">
   506→      <label>
   507→        Company website
   508→        <input
   509→          type="text"
   510→          name="website"
   511→          tabIndex={-1}
   512→          autoComplete="off"
   513→          value={honeypot}
   514→          onChange={(e) => setHoneypot(e.target.value)}
   515→        />
   516→      </label>
   517→    </div>
   518→  );
   519→
   520→  const consentCopy = (
   521→    <p
   522→      className={cn(
   523→        'text-xs leading-relaxed',
   524→        dark ? 'text-slate-400' : 'text-muted-foreground'
   525→      )}
   526→    >
   527→      By submitting, you agree to our{' '}
   528→      <Link
   529→        href="/privacy-policy"
   530→        className="font-medium text-purple-700 underline decoration-purple-300 underline-offset-2 hover:text-purple-800"
   531→      >
   532→        Privacy Policy
   533→      </Link>
   534→      . We reply within one business day and never share your details.
   535→    </p>
   536→  );
   537→
   538→  const submitButton = (
   539→    <button
   540→      type="submit"
   541→      disabled={submitting}
   542→      className="btn-primary-pill w-full disabled:pointer-events-none disabled:opacity-60"
   543→    >
   544→      {submitting ? (
   545→        <>
   546→          <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   547→          Sending…
   548→        </>
   549→      ) : (
   550→        <>
   551→          <Send className="h-4 w-4" aria-hidden="true" />
   552→          Send message
   553→        </>
   554→      )}
   555→    </button>
   556→  );
   557→
   558→  /* -------------------------- success ----------------------------- */
   559→
   560→  if (submitted) {
   561→    if (compact) {
   562→      return (
   563→        <div
   564→          className={cn('card-soft overflow-hidden rounded-[24px]', className)}
   565→          data-slot="lead-form-success"
   566→        >
   567→          <SuccessPanel leadId={leadId} dark={dark} onReset={resetToForm} />
   568→        </div>
   569→      );
   570→    }
   571→    if (bare) {
   572→      return (
   573→        <div
   574→          className={cn('relative rounded-[24px] bg-white p-6 sm:p-8 lg:p-10', className)}
   575→          data-slot="lead-form-success"
   576→        >
   577→          <SuccessPanel leadId={leadId} dark={dark} onReset={resetToForm} />
   578→        </div>
   579→      );
   580→    }
   581→    return (
   582→      <div
   583→        className={cn(wrapperClass, className)}
   584→        data-slot="lead-form-success"
   585→      >
   586→        <LeadRail />
   587→        <div className="relative bg-white p-6 sm:p-8 lg:p-10">
   588→          <SuccessPanel leadId={leadId} dark={dark} onReset={resetToForm} />
   589→        </div>
   590→      </div>
   591→    );
   592→  }
   593→
   594→  /* ------------------------------ bare ----------------------------- */
   595→  /*
   596→   * `bare` — the CLEAR single-column form used inside the contact page's
   597→   * gradient-border card. No dark rail, no nested splits: one white,
   598→   * airy, readable form with its own heading. Parent supplies the card.
   599→   */
   600→  if (bare) {
   601→    return (
   602→      <div className={cn('relative bg-white p-6 sm:p-8 lg:p-10', className)} data-slot="lead-form">
   603→        {submitted ? (
   604→          <SuccessPanel leadId={leadId} dark={dark} onReset={resetToForm} />
   605→        ) : (
   606→          <>
