    74→/**
    75→ * Splits the H1 so the service keyword phrase carries the brand gradient.
    76→ * Tries the full service name, then progressively shorter prefixes (min. two
    77→ * words) so phrases like "Website Maintenance & Support" still match their
    78→ * mention inside the hero title. Falls back to the first up-to-3 words.
    79→ * The returned tuple keeps the title text byte-for-byte intact.
    80→ */
    81→function splitHeroTitle(title: string, phrase: string): [string, string, string] {
    82→  const lower = title.toLowerCase();
    83→  const phraseWords = phrase.split(' ');
    84→  for (let len = Math.min(phraseWords.length, 4); len >= 2; len -= 1) {
    85→    const candidate = phraseWords.slice(0, len).join(' ').toLowerCase();
    86→    const idx = lower.indexOf(candidate);
    87→    if (idx === -1) continue;
    88→    const end = idx + candidate.length;
    89→    const boundaryBefore = idx === 0 || lower[idx - 1] === ' ';
    90→    const boundaryAfter = end === lower.length || lower[end] === ' ';
    91→    if (boundaryBefore && boundaryAfter) {
    92→      return [title.slice(0, idx), title.slice(idx, end), title.slice(end)];
    93→    }
    94→  }
    95→  const words = title.split(' ');
    96→  const head = words.slice(0, Math.min(3, words.length)).join(' ');
    97→  const tail = words.length > 3 ? ` ${words.slice(3).join(' ')}` : '';
    98→  return ['', head, tail];
    99→}
