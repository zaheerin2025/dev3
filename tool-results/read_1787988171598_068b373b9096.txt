   320→   880→interface MockupOpts {
   321→   881→  frame: string; bgMode: 'transparent' | 'solid' | 'gradient'; bg1: string; bg2: string;
   322→   882→  shadow: boolean; notch: boolean; scale: number;

   324→   884→
   325→   885→const MOCKUP_FRAMES = [
   326→   886→  { value: '#1c1c1e', label: 'Graphite' },

   333→   893→
   334→   894→function drawMockup(canvas: HTMLCanvasElement, img: HTMLImageElement, o: MockupOpts) {
   335→   895→  const s = o.scale;
   336→   896→  const bodyH = 1400 * s;
   337→   897→  const bezel = Math.max(12 * s, bodyH * 0.024);
   338→   898→  const aspect = clamp((img.naturalWidth || 1) / (img.naturalHeight || 1), 0.45, 0.88);
   339→   899→  const screenH = bodyH - bezel * 2;
   340→   900→  const screenW = screenH * aspect;
   341→   901→  const bodyW = screenW + bezel * 2;
   342→   902→  const margin = (o.shadow ? bodyH * 0.075 : bodyH * 0.03);
   343→   903→
   344→   904→  canvas.width = Math.round(bodyW + margin * 2);
   345→   905→  canvas.height = Math.round(bodyH + margin * 2);
   346→   906→  const ctx = canvas.getContext('2d');
   347→   907→  if (!ctx) return;
   348→   908→
   349→   909→  if (o.bgMode === 'solid') {
   350→   910→    ctx.fillStyle = hexToRgb(o.bg1) ? o.bg1 : '#f4f1fa';
   351→   911→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   352→   912→  } else if (o.bgMode === 'gradient') {
   353→   913→    const grad = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
   354→   914→    grad.addColorStop(0, hexToRgb(o.bg1) ? o.bg1 : '#7c3aed');
   355→   915→    grad.addColorStop(1, hexToRgb(o.bg2) ? o.bg2 : '#ec4899');
   356→   916→    ctx.fillStyle = grad;
   357→   917→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   358→   918→  }
   359→   919→
   360→   920→  const bx = margin, by = margin;
